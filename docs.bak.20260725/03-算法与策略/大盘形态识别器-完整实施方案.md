## 二、新增字段设计

### 2.1 新增字段清单

| 字段名 | 类型 | 说明 | 存储位置 |
|--------|------|------|---------|
| `mkt_shape` | VARCHAR(20) | 大盘形态 | `ext_indicators` JSON |
| `mkt_shape_detail` | VARCHAR(100) | 形态详细说明 | `ext_indicators` JSON |

### 2.2 5种形态定义

| 形态 | 判定条件 | 形状 | 交易含义 |
|------|---------|------|---------|
| **单边上行** | `low_pos < 0.2` AND `cur > 0` AND `drawdown < sig` | ↗ | 强势做多 |
| **单边下行** | `high_pos < 0.2` AND `cur < 0` AND `recovery < sig` | ↘ | 空仓观望 |
| **低开高走** | `low_pos > 0.2` AND `recovery > sig` AND `cur > 0` | ✓ | 洗盘完成，做多 |
| **高开低走** | `high_pos > 0.2` AND `drawdown > sig` AND `cur < 0` | ✗ | 诱多出货，观望 |
| **横盘** | `total_range < 0.5%` OR (其他条件都不满足) | — | 震荡观望 |

**阈值计算**：
```python
sig = total_range * 0.5  # 显著 = 振幅 × 0.5
```

**判定优先级**（高→低）：
1. 横盘（振幅<0.5%）
2. 单边上行/下行（极值在开头）
3. 低开高走/高开低走（极值在中后段）

```python
def compute_mkt_shape(mkt_vs_open_pct: float, time_full: str, current_date: str) -> dict:
    """
    计算大盘形态（实时）
    
    Returns:
        {'mkt_shape': str, 'mkt_shape_detail': str}
    """
    global _mkt_shape_date, _mkt_shape_history
    
    # 日期切换重置
    if _mkt_shape_date != current_date:
        _mkt_shape_history = []
        _mkt_shape_date = current_date
    
    # 添加当前点
    current_seconds = _time_to_seconds(time_full)
    _mkt_shape_history.append({
        'time': time_full,
        'time_sec': current_seconds,
        'vs_open': mkt_vs_open_pct
    })
    
    # 至少需要10个点才能判断形态（30秒）
    if len(_mkt_shape_history) < 10:
        return {'mkt_shape': '横盘', 'mkt_shape_detail': '数据不足，默认横盘'}
    
    # 计算结构指标
    vs_values = [h['vs_open'] for h in _mkt_shape_history]
    times = [h['time_sec'] for h in _mkt_shape_history]
    n = len(vs_values)
    
    cur_pct = vs_values[-1]
    high_pct = max(vs_values)
    low_pct = min(vs_values)
    total_range = high_pct - low_pct
    
    high_idx = vs_values.index(high_pct)
    low_idx = vs_values.index(low_pct)
    high_pos = high_idx / (n - 1) if n > 1 else 0
    low_pos = low_idx / (n - 1) if n > 1 else 0
    
    recovery = cur_pct - low_pct
    drawdown = high_pct - cur_pct
    
    # 阈值
    sig = total_range * 0.5 if total_range > 0 else 0.001
    
    # 判定（优先级从高到低）
    shape = '横盘'
    detail = ''
    
    # 1. 横盘
    if total_range < 0.5:
        shape = '横盘'
        detail = f'振幅{total_range:.2f}%<0.5%，波动过小'
    
    # 2. 单边上行
    elif low_pos < 0.2 and cur_pct > 0 and drawdown < sig:
        shape = '单边上行'
        detail = f'开盘即最低，当前{cur_pct:+.2f}%，回落{drawdown:.2f}%<阈值{sig:.2f}%'
    
    # 3. 单边下行
    elif high_pos < 0.2 and cur_pct < 0 and recovery < sig:
        shape = '单边下行'
        detail = f'开盘即最高，当前{cur_pct:+.2f}%，回升{recovery:.2f}%<阈值{sig:.2f}%'
    
    # 4. 低开高走
    elif low_pos > 0.2 and recovery > sig and cur_pct > 0:
        shape = '低开高走'
        detail = f'低点在{low_pos:.0%}位置，回升{recovery:.2f}%>{sig:.2f}%，当前{cur_pct:+.2f}%'
    
    # 5. 高开低走
    elif high_pos > 0.2 and drawdown > sig and cur_pct < 0:
        shape = '高开低走'
        detail = f'高点在{high_pos:.0%}位置，回落{drawdown:.2f}%>{sig:.2f}%，当前{cur_pct:+.2f}%'
    
    # 6. 其他情况归为横盘
    else:
        shape = '横盘'
        detail = f'高{high_pct:+.2f}%@{high_pos:.0%} 低{low_pct:+.2f}%@{low_pos:.0%} 收{cur_pct:+.2f}%，无明确形态'
    
    return {
        'mkt_shape': shape,
        'mkt_shape_detail': detail
    }
```

### 3.3 detect_market_shape.py 修改

简化为提供计算函数，供monitor_bond.py调用：

```python
#!/usr/bin/env python3
"""
大盘日内走势形态识别器 - 核心计算模块

提供形态计算函数，供monitor_bond.py实时调用

5种形态：
    单边上行 / 单边下行 / 低开高走 / 高开低走 / 横盘
"""

import numpy as np
from typing import List, Dict, Tuple

# 形态配置
SHAPE_CONFIG = {
    'flat_range': 0.5,      # 横盘阈值
    'significant_ratio': 0.5,  # 显著比例
}


def calc_mkt_shape(vs_open_history: List[Dict]) -> Tuple[str, str]:
    """
    计算大盘形态（核心函数）
    
    Args:
        vs_open_history: [{'time': str, 'vs_open': float}, ...]
    
    Returns:
        (形态名, 详细说明)
    """
    if len(vs_open_history) < 5:
        return '横盘', '数据不足'
    
    vs_values = [h['vs_open'] for h in vs_open_history]
    cur_pct = vs_values[-1]
    high_pct = max(vs_values)
    low_pct = min(vs_values)
    total_range = high_pct - low_pct
    
    n = len(vs_values)
    high_idx = vs_values.index(high_pct)
    low_idx = vs_values.index(low_pct)
    high_pos = high_idx / (n - 1)
    low_pos = low_idx / (n - 1)
    
    recovery = cur_pct - low_pct
    drawdown = high_pct - cur_pct
    sig = total_range * SHAPE_CONFIG['significant_ratio']
    
    # 判定
    if total_range < SHAPE_CONFIG['flat_range']:
        return '横盘', f'振幅{total_range:.2f}%<0.5%'
    
    if low_pos < 0.2 and cur_pct > 0 and drawdown < sig:
        return '单边上行', f'开盘即最低，当前{cur_pct:+.2f}%'
    
    if high_pos < 0.2 and cur_pct < 0 and recovery < sig:
        return '单边下行', f'开盘即最高，当前{cur_pct:+.2f}%'
    
    if low_pos > 0.2 and recovery > sig and cur_pct > 0:
        return '低开高走', f'低点在{low_pos:.0%}，回升{recovery:.2f}%'
    
    if high_pos > 0.2 and drawdown > sig and cur_pct < 0:
        return '高开低走', f'高点在{high_pos:.0%}，回落{drawdown:.2f}%'
    
    return '横盘', f'无明确形态'


def main():
    """命令行测试工具"""
    import argparse
    import pandas as pd
    from sqlalchemy import text
    from gs2026.utils import config_util
    
    parser = argparse.ArgumentParser()
    parser.add_argument('date', help='日期 YYYYMMDD')
    parser.add_argument('--time', help='截止时间 HH:MM')
    args = parser.parse_args()
    
    # 加载数据
    engine = config_util.get_engine()
    table = f"monitor_zq_sssj_{args.date}"
    
    with engine.connect() as conn:
        # 取一只债券的全天数据
        code_df = pd.read_sql(text(f"SELECT bond_code FROM {table} LIMIT 1"), conn)
        if code_df.empty:
            print("无数据")
            return
        code = code_df['bond_code'].iloc[0]
        
        sql = f"SELECT time, ext_indicators FROM {table} WHERE bond_code='{code}' ORDER BY time"
        if args.time:
            sql = f"SELECT time, ext_indicators FROM {table} WHERE bond_code='{code}' AND time <= '{args.time.replace(':','')}' ORDER BY time"
        
        df = pd.read_sql(text(sql), conn)
    
    # 解析mkt_vs_open_pct
    import json
    history = []
    for _, row in df.iterrows():
        ext = json.loads(row['ext_indicators']) if row['ext_indicators'] else {}
        vs_open = ext.get('mkt_vs_open_pct')
        if vs_open is not None:
            history.append({'time': row['time'], 'vs_open': vs_open})
    
    # 计算形态
    shape, detail = calc_mkt_shape(history)
    print(f"日期: {args.date} 时间: {args.time or '收盘'}")
    print(f"【形态判定】{shape}")
    print(f"  说明: {detail}")
    print(f"  数据点数: {len(history)}")


if __name__ == '__main__':
    main()
```

## 五、延迟分析

| 延迟来源 | 数值 | 说明 |
|---------|------|------|
| 数据源 | 3秒 | 每tick更新 |
| 形态计算 | <5ms | 简单数学运算 |
| 历史存储 | O(1) | deque自动维护 |
| **总延迟** | **~3秒** | 可接受 |

**形态变化延迟**：
- 从"单边下行"变"V型"需要：创出新低 → 回升显著（约10-30分钟）
- 从"V型"变"单边上行"需要：突破前高（约10-30分钟）
