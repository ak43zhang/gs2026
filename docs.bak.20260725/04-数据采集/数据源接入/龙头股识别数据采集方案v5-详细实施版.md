```python
def get_effective_trade_date(target_date: str = None) -> str:
    """
    获取有效的交易日期
    - 如果target_date是交易日，返回target_date
    - 如果不是交易日，返回上一个交易日
    
    Args:
        target_date: 目标日期(YYYYMMDD)，None则使用当天
        
    Returns:
        有效的交易日期(YYYYMMDD)
    """
    if target_date is None:
        target_date = time.strftime("%Y%m%d", time.localtime())
    
    # 查询data_jyrl表判断是否是交易日
    sql = """
        SELECT trade_date 
        FROM data_jyrl 
        WHERE trade_date <= :target_date AND trade_status = '1'
        ORDER BY trade_date DESC 
        LIMIT 1
    """
    
    with engine.connect() as conn:
        result = conn.execute(text(sql), {'target_date': target_date})
        row = result.fetchone()
        
        if row:
            return row[0]
        else:
            # 如果没有找到，返回target_date（兜底）
            return target_date
```

```python
def stock_spot_em() -> None:
    """
    采集东方财富A股实时行情（含流通市值）
    位置：写在industry_code_component_ths()后面（约385行）
    
    逻辑：
    1. 获取有效的交易日期（当天或上一交易日）
    2. 先删除该日期的数据（保证幂等性）
    3. 采集akshare数据
    4. 写入MySQL（表自动生成）
    5. 创建索引（如果不存在）
    """
    import akshare as ak
    from sqlalchemy import text
    
    table_name = "data_stock_spot_em"
    
    # 1. 获取有效的交易日期
    trade_date = get_effective_trade_date()
    logger.info(f"[stock_spot_em] 有效交易日期: {trade_date}")
    
    try:
        # 2. 先删除该日期的数据（幂等性）
        if mysql_tool.check_table_exists(table_name):
            delete_sql = f"DELETE FROM `{table_name}` WHERE `trade_date` = '{trade_date}'"
            mysql_tool.delete_data(delete_sql)
            logger.info(f"[stock_spot_em] 已删除 {trade_date} 的旧数据")
        
        # 3. 采集akshare数据
        logger.info("[stock_spot_em] 开始采集A股实时行情...")
        df = ak.stock_zh_a_spot_em()
        
        if df.empty:

            return
        
        logger.info(f"[stock_spot_em] 采集到 {df.shape[0]} 条数据")
        
        # 4. 字段映射（标准化字段名）
        column_mapping = {
            '序号': 'seq_no',
            '代码': 'stock_code',
            '名称': 'stock_name',
            '最新价': 'price',
            '涨跌幅': 'change_pct',
            '涨跌额': 'change_amount',
            '成交量': 'volume',
            '成交额': 'amount',
            '振幅': 'amplitude',
            '最高': 'high',
            '最低': 'low',
            '今开': 'open_price',
            '昨收': 'prev_close',
            '量比': 'volume_ratio',
            '换手率': 'turnover_rate',
            '市盈率-动态': 'pe_ratio',
            '市净率': 'pb_ratio',
            '总市值': 'total_market_cap',
            '流通市值': 'circulating_market_cap',  # 关键字段
            '涨速': 'rise_speed',
            '5分钟涨跌': 'change_5min',
            '60日涨跌幅': 'change_60d',
            '年初至今涨跌幅': 'change_ytd',
            '所属行业': 'industry'
        }
        
        # 重命名列（只保留存在的列）
        existing_columns = {k: v for k, v in column_mapping.items() if k in df.columns}
        df = df.rename(columns=existing_columns)
        
        # 5. 数据清洗
        # 转换数值字段
        numeric_columns = [
            'price', 'change_pct', 'change_amount', 'volume', 'amount',
            'amplitude', 'high', 'low', 'open_price', 'prev_close',
            'volume_ratio', 'turnover_rate', 'pe_ratio', 'pb_ratio',
            'total_market_cap', 'circulating_market_cap', 'rise_speed',
            'change_5min', 'change_60d', 'change_ytd'
        ]
        
        for col in numeric_columns:
            if col in df.columns:
                df[col] = pd.to_numeric(df[col], errors='coerce')
        
        # 6. 添加交易日期
        df['trade_date'] = trade_date
        
        # 7. 写入MySQL（表自动生成）
        with engine.begin() as conn:
            df.to_sql(
                name=table_name, 
                con=conn, 
                if_exists='append', 
                index=False,
                chunksize=1000  # 分批写入，避免大数据量问题
            )
            logger.info(f"[stock_spot_em] 表名：{table_name}、日期：{trade_date}、数量：{df.shape[0]}")
        
        # 8. 创建索引（如果不存在）
        _create_stock_spot_em_indexes()
        
    except Exception as e:
        logger.error(f"[stock_spot_em] {trade_date} 异常：{e}")
        import traceback
        logger.error(traceback.format_exc())


def _create_stock_spot_em_indexes():
    """
    为data_stock_spot_em表创建索引（如果不存在）
    """
    table_name = "data_stock_spot_em"
    
    indexes = [
        ("idx_trade_date", "trade_date"),
        ("idx_stock_code", "stock_code"),
        ("idx_circulating_cap", "circulating_market_cap"),
        ("idx_trade_date_code", "trade_date, stock_code"),
    ]
    
    with engine.connect() as conn:
        for idx_name, idx_columns in indexes:
            try:
                # 检查索引是否存在
                check_sql = f"""
                    SELECT COUNT(*) FROM information_schema.STATISTICS 
                    WHERE TABLE_SCHEMA = DATABASE() 
                    AND TABLE_NAME = '{table_name}' 
                    AND INDEX_NAME = '{idx_name}'
                """
                result = conn.execute(text(check_sql))
                count = result.scalar()
                
                if count == 0:
                    # 创建索引
                    create_sql = f"CREATE INDEX `{idx_name}` ON `{table_name}` ({idx_columns})"
                    conn.execute(text(create_sql))
                    logger.info(f"[stock_spot_em] 创建索引：{idx_name} ({idx_columns})")
            except Exception as e:
                logger.warning(f"[stock_spot_em] 创建索引 {idx_name} 失败：{e}")
```

## 四、前端实时计算函数

```javascript
/**
 * 获取龙头展示role（前端实时计算）
 * 
 * @param {Object} stock - 股票对象
 * @param {Array} mainlineStocks - 同主线所有股票
 * @param {String} tradeDate - 交易日期YYYYMMDD
 * @returns {Object} - {displayRole, leaderType, marketCap}
 */
async function getLeaderDisplayRole(stock, mainlineStocks, tradeDate) {
    const result = {
        originalRole: stock.role,
        displayRole: stock.role,
        leaderType: null,  // 权重龙头/板块中军/小盘情绪龙
        marketCap: null,
        isEstimated: false
    };
    
    // 1. 如果AI已定义细分类型，直接返回
    if (['权重龙头', '板块中军', '小盘情绪龙'].includes(stock.role)) {
        result.leaderType = stock.role;
        return result;
    }
    
    // 2. 如果AI只定义了'龙头'，需要根据市值细分
    if (stock.role === '龙头') {
        // 查询流通市值
        const marketCap = await queryMarketCap(stock.stock_code, tradeDate);
        result.marketCap = marketCap;
        
        if (marketCap) {
            if (marketCap > 500) {
                result.leaderType = '权重龙头';
            } else if (marketCap >= 200) {
                result.leaderType = '板块中军';
            } else {
                result.leaderType = '小盘情绪龙';
            }
            result.displayRole = result.leaderType + '(细分)';
            result.isEstimated = true;
        } else {
            // 无市值数据，保持原样
            result.leaderType = '龙头';
        }
    }
    
    // 3. 如果AI未定义龙头，检查是否需要按时间补龙头
    // （这部分逻辑根据您的需求决定是否启用）
    const hasAiLeader = mainlineStocks.some(s => 
        ['龙头', '权重龙头', '板块中军', '小盘情绪龙'].includes(s.role)
    );
    
    if (!hasAiLeader && stock.role !== '龙头') {
        // AI未定义龙头，按时间排序看是否是第1只
        const sortedStocks = mainlineStocks.sort((a, b) => 
            a.anomaly_time.localeCompare(b.anomaly_time)
        );
        const rank = sortedStocks.findIndex(s => s.stock_code === stock.stock_code);
        
        if (rank === 0) {
            // 第1只，查询市值判断类型
            const marketCap = await queryMarketCap(stock.stock_code, tradeDate);
            result.marketCap = marketCap;
            
            if (marketCap) {
                if (marketCap > 500) {
                    result.leaderType = '权重龙头';
                } else if (marketCap >= 200) {
                    result.leaderType = '板块中军';
                } else {
                    result.leaderType = '小盘情绪龙';
                }
                result.displayRole = result.leaderType + '(估算)';
                result.isEstimated = true;
            }
        }
    }
    
    return result;
}

/**
 * 查询流通市值
 */
async function queryMarketCap(stockCode, tradeDate) {
    try {
        const response = await fetch(`/api/stock/market-cap?code=${stockCode}&date=${tradeDate}`);
        const data = await response.json();
        return data.circulating_market_cap;
    } catch (e) {
        console.error('查询流通市值失败:', e);
        return null;
    }
}
```

```python
from flask import Blueprint, request, jsonify
from sqlalchemy import text

api_bp = Blueprint('api', __name__)

@api_bp.route('/stock/market-cap', methods=['GET'])
def get_stock_market_cap():
    """
    查询股票流通市值
    
    参数:
        code: 股票代码
        date: 交易日期(YYYYMMDD)，可选，默认最新
    """
    stock_code = request.args.get('code')
    trade_date = request.args.get('date')
    
    if not stock_code:
        return jsonify({'error': '缺少股票代码'}), 400
    
    try:
        if trade_date:
            # 查询指定日期
            sql = """
                SELECT circulating_market_cap, total_market_cap
                FROM data_stock_spot_em
                WHERE stock_code = :code AND trade_date = :date
                LIMIT 1
            """
            result = db.session.execute(text(sql), {
                'code': stock_code,
                'date': trade_date
            })
        else:
            # 查询最新日期
            sql = """
                SELECT circulating_market_cap, total_market_cap, trade_date
                FROM data_stock_spot_em
                WHERE stock_code = :code
                ORDER BY trade_date DESC
                LIMIT 1
            """
            result = db.session.execute(text(sql), {'code': stock_code})
        
        row = result.fetchone()
        
        if row:
            return jsonify({
                'stock_code': stock_code,
                'circulating_market_cap': float(row[0]) if row[0] else None,
                'total_market_cap': float(row[1]) if row[1] else None,
                'trade_date': row[2] if len(row) > 2 else trade_date
            })
        else:
            return jsonify({'error': '未找到数据'}), 404
            
    except Exception as e:
        return jsonify({'error': str(e)}), 500
```
