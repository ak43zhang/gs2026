## 四、前端渲染逻辑

```javascript
// 板块消息渲染
function renderSectorMsg(sector_msg) {
    if (!sector_msg || sector_msg.length === 0) return '<p>无板块消息</p>';
    
    return sector_msg.map(s => `
        <div class="msg-group">
            <h4>🏭 ${s.板块}</h4>
            <ul>
                ${(s.板块相关消息 || []).map(m => `<li>${m}</li>`).join('')}
            </ul>
        </div>
    `).join('');
}

// 概念消息渲染
function renderConceptMsg(concept_msg) {
    if (!concept_msg || concept_msg.length === 0) return '<p>无概念消息</p>';
    
    return concept_msg.map(c => `
        <div class="msg-group">
            <h4>💡 ${c.概念}</h4>
            <ul>
                ${(c.概念相关消息 || []).map(m => `<li>${m}</li>`).join('')}
            </ul>
        </div>
    `).join('');
}

// 龙头股消息渲染
function renderLeadingStockMsg(leading_stock_msg) {
    if (!leading_stock_msg || leading_stock_msg.length === 0) return '<p>无龙头股消息</p>';
    
    return leading_stock_msg.map((l, idx) => `
        <div class="msg-group">
            <h4>👑 ${l.龙头股} ${idx === 0 ? '（领涨）' : ''}</h4>
            <ul>
                ${(l.龙头股相关消息 || []).map(m => `<li>${m}</li>`).join('')}
            </ul>
        </div>
    `).join('');
}

// 影响消息渲染
function renderInfluenceMsg(influence_msg) {
    if (!influence_msg || influence_msg.length === 0) return '<p>无影响消息</p>';
    
    // 按时间倒序
    const sorted = [...influence_msg].sort((a, b) => 
        new Date(b.影响消息时间) - new Date(a.影响消息时间)
    );
    
    return sorted.map(i => `
        <div class="influence-item">
            <div class="influence-date">📅 ${i.影响消息时间}</div>
            <div class="influence-content">${i.影响消息}</div>
        </div>
    `).join('');
}

// 预期消息渲染
function renderExpectMsg(expect_msg) {
    if (!expect_msg || expect_msg.length === 0) return '<p>无预期消息</p>';
    
    // 按时间倒序
    const sorted = [...expect_msg].sort((a, b) => 
        new Date(b.预期消息时间) - new Date(a.预期消息时间)
    );
    
    return sorted.map(e => `
        <div class="expect-item">
            <div class="expect-header">
                <span class="expect-date">📅 ${e.预期消息时间}</span>
                <span class="expect-continuity ${e.延续性 === '是' ? 'yes' : 'no'}">
                    ${e.延续性 === '是' ? '✅ 延续' : '❌ 不延续'}
                </span>
            </div>
            <div class="expect-content">${e.预期消息}</div>
        </div>
    `).join('');
}

// 深度分析渲染
function renderDeepAnalysis(deep_analysis) {
    if (!deep_analysis || deep_analysis.length === 0) return '<p>无深度分析</p>';
    
    const iconMap = {
        '技术面': '📈',
        '基本面': '📊',
        '资金面': '💰',
        '情绪面': '😊'
    };
    
    return deep_analysis.map(d => {
        const parts = d.split(':');
        const title = parts[0].trim();
        const content = parts.slice(1).join(':').trim();
        const icon = iconMap[title] || '•';
        
        // 将内容按逗号分割成多条
        const points = content.split(/[,，]/).filter(p => p.trim());
        
        return `
            <div class="analysis-block">
                <h4>${icon} ${title}</h4>
                <ul>
                    ${points.map(p => `<li>${p.trim()}</li>`).join('')}
                </ul>
            </div>
        `;
    }).join('');
}
```

## 四、前端渲染逻辑

```javascript
// 板块消息渲染
function renderSectorMsg(sector_msg) {
    if (!sector_msg || sector_msg.length === 0) return '<p>无板块消息</p>';
    
    return sector_msg.map(s => `
        <div class="msg-group">
            <h4>🏭 ${s.板块}</h4>
            <ul>
                ${(s.板块相关消息 || []).map(m => `<li>${m}</li>`).join('')}
            </ul>
        </div>
    `).join('');
}

// 概念消息渲染
function renderConceptMsg(concept_msg) {
    if (!concept_msg || concept_msg.length === 0) return '<p>无概念消息</p>';
    
    return concept_msg.map(c => `
        <div class="msg-group">
            <h4>💡 ${c.概念}</h4>
            <ul>
                ${(c.概念相关消息 || []).map(m => `<li>${m}</li>`).join('')}
            </ul>
        </div>
    `).join('');
}

// 龙头股消息渲染
function renderLeadingStockMsg(leading_stock_msg) {
    if (!leading_stock_msg || leading_stock_msg.length === 0) return '<p>无龙头股消息</p>';
    
    return leading_stock_msg.map((l, idx) => `
        <div class="msg-group">
            <h4>👑 ${l.龙头股} ${idx === 0 ? '（领涨）' : ''}</h4>
            <ul>
                ${(l.龙头股相关消息 || []).map(m => `<li>${m}</li>`).join('')}
            </ul>
        </div>
    `).join('');
}

// 影响消息渲染
function renderInfluenceMsg(influence_msg) {
    if (!influence_msg || influence_msg.length === 0) return '<p>无影响消息</p>';
    
    // 按时间倒序
    const sorted = [...influence_msg].sort((a, b) => 
        new Date(b.影响消息时间) - new Date(a.影响消息时间)
    );
    
    return sorted.map(i => `
        <div class="influence-item">
            <div class="influence-date">📅 ${i.影响消息时间}</div>
            <div class="influence-content">${i.影响消息}</div>
        </div>
    `).join('');
}

// 预期消息渲染
function renderExpectMsg(expect_msg) {
    if (!expect_msg || expect_msg.length === 0) return '<p>无预期消息</p>';
    
    // 按时间倒序
    const sorted = [...expect_msg].sort((a, b) => 
        new Date(b.预期消息时间) - new Date(a.预期消息时间)
    );
    
    return sorted.map(e => `
        <div class="expect-item">
            <div class="expect-header">
                <span class="expect-date">📅 ${e.预期消息时间}</span>
                <span class="expect-continuity ${e.延续性 === '是' ? 'yes' : 'no'}">
                    ${e.延续性 === '是' ? '✅ 延续' : '❌ 不延续'}
                </span>
            </div>
            <div class="expect-content">${e.预期消息}</div>
        </div>
    `).join('');
}

// 深度分析渲染
function renderDeepAnalysis(deep_analysis) {
    if (!deep_analysis || deep_analysis.length === 0) return '<p>无深度分析</p>';
    
    const iconMap = {
        '技术面': '📈',
        '基本面': '📊',
        '资金面': '💰',
        '情绪面': '😊'
    };
    
    return deep_analysis.map(d => {
        const parts = d.split(':');
        const title = parts[0].trim();
        const content = parts.slice(1).join(':').trim();
        const icon = iconMap[title] || '•';
        
        // 将内容按逗号分割成多条
        const points = content.split(/[,，]/).filter(p => p.trim());
        
        return `
            <div class="analysis-block">
                <h4>${icon} ${title}</h4>
                <ul>
                    ${points.map(p => `<li>${p.trim()}</li>`).join('')}
                </ul>
            </div>
        `;
    }).join('');
}
```

## 2. 涨停查询性能分析


## 当前实现分析

### 查询流程

```
1. 从宽表缓存获取所有有选中标签的股票
   - 遍历所有股票（约5000只）
   - 检查是否匹配标签
   - 假设匹配1000只
   
2. 查询这1000只股票在指定时间点的价格
   a. 先尝试Redis
      - 获取整个时间点的所有股票数据（可能5000+条记录）
      - 遍历筛选出匹配的1000只
   b. Redis失败则查询MySQL
      - SQL: SELECT ... WHERE stock_code IN ('000001','000002',...1000个)
      - 大IN列表可能导致性能问题
      
3. 查询转债价格
   - 同样的问题：大IN列表
   
4. 组装结果
   - 遍历1000只股票
```

### 性能瓶颈

**瓶颈1：Redis查询**
```python
# 获取整个时间点的所有数据
df = pd.DataFrame(json.loads(data))  # 可能5000+条
for _, row in df.iterrows():         # 遍历5000+
    if row['stock_code'] in stock_codes:  # 1000次匹配
        result[row['stock_code']] = {...}
```
问题：即使只需要1000只，也要加载和遍历5000+条

**瓶颈2：MySQL大IN列表**
```python
placeholders = ','.join([f"'{code}'" for code in stock_codes])  # 1000个
sql = f"... WHERE stock_code IN ({placeholders})"
```
问题：IN列表太长，MySQL优化器可能选择全表扫描

**瓶颈3：多次Redis连接**
```python
# 股票价格查询
redis_client = _get_redis_client()  # 第1次

# 转债价格查询  
redis_client = _get_redis_client()  # 第2次
```

## 优化方案

### 方案A：批量查询优化（推荐）

**优化1：Redis使用批量获取**
```python
# 改为批量获取指定股票
for code in stock_codes:
    redis_key = f"monitor_gp_sssj_{date_str}:{time}:{code}"
    data = redis_client.get(redis_key)
    if data:
        result[code] = json.loads(data)
```
但当前Redis数据结构是整个DataFrame，不是单只股票。

**优化2：MySQL分批查询**
```python
# 将1000只股票分批，每批100只
batch_size = 100
for i in range(0, len(stock_codes), batch_size):
    batch = stock_codes[i:i+batch_size]
    placeholders = ','.join([f"'{code}'" for code in batch])
    sql = f"... WHERE stock_code IN ({placeholders})"
    # 查询并合并结果
```

**优化3：使用JOIN替代大IN**
```python
# 创建临时表或子查询
sql = f"""
    SELECT s.stock_code, s.short_name, s.price, s.change_pct
    FROM {table_name} s
    INNER JOIN (
        SELECT '{stock_codes[0]}' as code
        UNION ALL SELECT '{stock_codes[1]}'
        ...
    ) codes ON s.stock_code = codes.code
    WHERE s.time = '{time}'
"""
```

## 3. 涨停查询最终方案


## 设计思路理解

### 涨停行概选股的完整流程

```
1. 获取涨停股票（指定时间点）
   ↓
2. 统计涨停股票的行业/概念频次
   ↓
3. 展示行业/概念标签（带频次）
   ↓
4. 用户选择标签（如"半导体"）
   ↓
5. 【关键】查询所有属于半导体的股票（不限涨停）
   ↓
6. 以涨停行概形式展示（分组、命中标签等）
```

### 关键理解

**行业/概念标签的数据来源：**
- 从涨停股票统计而来（所以频次正确）

**点击标签后的查询范围：**
- **所有有该行业/概念的股票**（不限于涨停）
- 不是只查询涨停股票中属于该行业的

**展示形式：**
- 与涨停行概相同（分组展示、命中标签高亮等）

## 当前问题

当前`filter_ztb_snapshot`实现：
```python
# 1. 获取涨停股票
df = _get_zt_stocks_at_time(date_str, time)  # 只获取涨停的

# 2. 筛选有选中标签的
for code in stock_codes:
    if selected_tags:
        if not matched:  # 不匹配就跳过
            continue
```

**问题：** 只查询了涨停股票中匹配标签的

## 正确方案

### 修改`filter_ztb_snapshot`

```python
def filter_ztb_snapshot(date: str, time: str, selected_tags: List[Dict], filters: Dict) -> Dict:
    """根据标签筛选股票（查询所有有标签的股票，不限涨停）"""
    
    # 【修改1】从宽表缓存获取所有有选中标签的股票（不限涨停）
    from gs2026.dashboard2.services.stock_picker_service import _stock_cache, load_memory_cache
    
    if not _stock_cache:
        load_memory_cache()
    
    selected_industries_set = set(t['name'] for t in selected_tags if t['type'] == 'industry')
    selected_concepts_set = set(t['name'] for t in selected_tags if t['type'] == 'concept')
    
    # 查询所有有选中标签的股票
    stock_matches = {}
    for stock_code, data in _stock_cache.items():
        all_tags = data['industries'] | data['concepts']
        matched = (selected_industries_set | selected_concepts_set) & all_tags
        
        if matched:
            matched_industries = list(selected_industries_set & data['industries'])
            matched_concepts = list(selected_concepts_set & data['concepts'])
            
            stock_matches[stock_code] = {
                'stock_name': data['stock_name'],
                'bond_code': data['bond_code'] or '-',
                'bond_name': data['bond_name'] or '-',
                'industries': matched_industries,
                'concepts': matched_concepts,
                'match_count': len(matched)
            }
    
    # 【修改2】查询这些股票在指定时间点的实时数据
    date_str = date.replace('-', '')
    all_codes = list(stock_matches.keys())
    price_data = _query_prices_at_time(all_codes, date_str, time)
    
    # 【修改3】查询转债价格
    bond_codes = [m['bond_code'] for m in stock_matches.values() if m['bond_code'] != '-']
    bond_price_data = _query_bond_prices_at_time(bond_codes, date_str, time)
    
    # 【修改4】组装结果（复用交叉选股逻辑）
    groups = defaultdict(list)
    with_bond_count = 0
    
    for stock_code, match_info in stock_matches.items():
        price_info = price_data.get(stock_code, {})
        bond_info = bond_price_data.get(match_info['bond_code'], {})
        
        # 获取该股票的所有行业（与选中标签无关）
        stock_data = _stock_cache.get(stock_code, {})
        all_industries = sorted(list(stock_data.get('industries', set()))) if stock_data else []
        
        stock_result = {
            'stock_code': stock_code,
            'stock_name': match_info['stock_name'] or price_info.get('short_name', ''),
            'change_pct': price_info.get('change_pct', 0),
            'price': price_info.get('price', 0),
            'bond_code': match_info['bond_code'],
            'bond_name': match_info['bond_name'],
            'bond_change_pct': bond_info.get('change_pct', 0),
            'industry_name': '、'.join(all_industries) if all_industries else '',
            'matched_industries': match_info['industries'],
            'matched_concepts': match_info['concepts'],
            'matched_tags_display': '\n'.join(match_info['industries'] + match_info['concepts'])
        }
        
        groups[match_info['match_count']].append(stock_result)
        
        if match_info['bond_code'] != '-':
            with_bond_count += 1
    
    # 【修改5】构建返回结果（与交叉选股一致）
    result_groups = []
    for count in sorted(groups.keys(), reverse=True):
        stocks = groups[count]
        stocks.sort(key=lambda x: x['change_pct'], reverse=True)
        
        if count == len(selected_tags):
            label = f"命中全部 {count} 个"
        else:
            label = f"命中 {count} 个"
        
        result_groups.append({
            'match_count': count,
            'label': label,
            'stocks': stocks
        })
    
    return {
        'tags': selected_tags,
        'groups': result_groups,
        'summary': {
            'total_stocks': len(stock_matches),
            'with_bond': with_bond_count,
            'query_time_ms': 0
        }
    }
```

```python
def _query_prices_at_time(stock_codes: List[str], date_str: str, time: str) -> Dict[str, Dict]:
    """查询指定时间点的股票价格"""
    if not stock_codes:
        return {}
    
    result = {}
    
    # 1. 尝试Redis
    try:
        from gs2026.utils.redis_util import _get_redis_client
        redis_client = _get_redis_client()
        
        if redis_client:
            redis_key = f"monitor_gp_sssj_{date_str}:{time}"
            data = redis_client.get(redis_key)
            if data:
                import json
                df = pd.DataFrame(json.loads(data))
                for _, row in df.iterrows():
                    if row['stock_code'] in stock_codes:
                        result[row['stock_code']] = {
                            'price': float(row.get('price', 0)),
                            'change_pct': float(row.get('change_pct', 0)),
                            'short_name': row.get('short_name', '')
                        }
                return result
    except Exception as e:
        logger.warning(f"Redis查询价格失败: {e}")
    
    # 2. 从MySQL
    try:
        table_name = f"monitor_gp_sssj_{date_str}"
        placeholders = ','.join([f"'{code}'" for code in stock_codes])
        sql = f"""
            SELECT stock_code, short_name, price, change_pct
            FROM {table_name}
            WHERE time = '{time}' AND stock_code IN ({placeholders})
        """
        df = pd.read_sql(sql, engine)
        for _, row in df.iterrows():
            result[row['stock_code']] = {
                'price': float(row['price']),
                'change_pct': float(row['change_pct']),
                'short_name': row['short_name']
            }
    except Exception as e:
        logger.error(f"MySQL查询价格失败: {e}")
    
    return result
```

## 6. 涨停查询业务分析


## 字段含义

### `is_zt` vs `ever_zt`

| 字段 | 含义 | 示例 |
|------|------|------|
| `is_zt` | **当前时刻是否涨停** | 09:30:00 is_zt=1, 09:30:15 is_zt=0（打开涨停） |
| `ever_zt` | **当天是否曾涨停** | 只要当天曾涨停，所有记录的ever_zt=1 |

## 两种查询方式的区别

### 方式A：`is_zt=1 AND time='指定时间'`

**含义：** 查询"指定时间点正在涨停"的股票

**适用场景：**
- 查看某个瞬间的涨停情况
- 了解当时市场上有哪些股票在涨停板上

**示例：**
- 09:30:00 查询 → 返回10只（当时正在涨停）
- 09:30:15 查询 → 返回8只（2只打开涨停）
- 14:30:00 查询 → 返回50只（下午涨停潮）

### 方式B：`ever_zt=1 AND time<='指定时间'`

**含义：** 查询"到指定时间点为止曾涨停"的股票

**适用场景：**
- 复盘分析：查看当天哪些股票曾涨停
- 统计涨停持续性

**示例：**
- 09:30:00 查询 → 返回10只
- 09:30:15 查询 → 返回10只（包含已打开的）
- 14:30:00 查询 → 返回100只（全天累计）

## 当前实现的问题

### 现状
- `get_ztb_snapshot` 使用 `ever_zt=1` → 获取曾涨停股票
- `filter_ztb_snapshot` 筛选时展示的是这些股票的**当前状态**

### 问题
用户期望的是：**查看指定时间点的瞬时涨停状态**

即：
1. 选择 09:30:00 → 看到当时正在涨停的股票及其当时的价格
2. 选择 14:30:00 → 看到当时正在涨停的股票及其当时的价格

## 正确的业务逻辑

### 场景：复盘指定时间点的涨停情况

**用户需求：** "我想看看14:30的时候，市场上有哪些股票在涨停，当时的价格是多少"

**正确实现：**
```sql
-- 查询指定时间点正在涨停的股票
SELECT * FROM monitor_gp_sssj_{date} 
WHERE time = '14:30:00' AND is_zt = 1

-- 获取这些股票的行业概念（从宽表）
-- 展示当时的涨跌幅、价格等
```

### 场景：查看当天所有曾涨停的股票

**用户需求：** "我想看看今天哪些股票曾涨停过"

**正确实现：**
```sql
-- 查询当天曾涨停的所有股票
SELECT DISTINCT stock_code FROM monitor_gp_sssj_{date} 
WHERE ever_zt = 1
```

## 建议方案

### 方案1：时间轴查询 `is_zt=1`（推荐）

**理由：**
1. 时间轴选择"14:30"，用户期望看到的是14:30那一刻的涨停情况
2. 与默认模式（最新时间涨停）逻辑一致，只是时间不同
3. 数据更准确，反映当时市场状态

**修改：**
```python
def _get_zt_stocks_at_time(date_str: str, time: str) -> pd.DataFrame:
    """获取指定时间点正在涨停的股票"""
    # 查询 is_zt=1 且 time=指定时间
    sql = f"""
        SELECT * FROM monitor_gp_sssj_{date_str}
        WHERE time = '{time}' AND is_zt = 1
    """
```

### 方案2：保持 `ever_zt=1`

**理由：**
1. 可以看到全天曾涨停的所有股票
2. 了解哪些股票"曾经强势"

**但：** 与时间轴选择时间点的交互不符

## 结论

**推荐方案1：使用 `is_zt=1 AND time='指定时间'`**

时间轴模式应该查询"指定时间点正在涨停"的股票，而不是"到该时间点为止曾涨停"的股票。

用户选择"14:30"，期望看到的是14:30那一刻的涨停情况，而不是全天累计。

### 方案A：添加Redis缓存（推荐）

#### 3.1.3 实现代码

```python
def get_ztb_list(...):
    # 生成缓存键
    cache_key = f"ztb:list:{date}:{page}:{page_size}:{hash(filters)}"
    
    # 尝试从缓存获取
    cached = redis.get(cache_key)
    if cached:
        return json.loads(cached)
    
    # 缓存未命中，查询数据库
    result = _query_from_db(...)
    
    # 写入缓存
    redis.setex(cache_key, LIST_TTL, json.dumps(result))
    
    return result
```

- 需要处理缓存一致性

```python
def get_total_count(trade_date, filters):
```

## 二、解决方案（方案A）

### 2.1 设计思路
在`result_processor.py`的`_extract_ztb_record`函数中：
1. 接收`stock_code`和`trade_date`参数
2. 查询`ztb_day`表获取首次涨停时间
3. 使用真实时间替换AI分析时间

#### 修改1：添加查询函数
```python
def _get_zt_time_from_db(stock_code: str, trade_date: str) -> Optional[str]:
    """从ztb_day查询首次涨停时间"""
    try:
        sql = f"""
            SELECT 首次涨停时间 
            FROM ztb_day 
            WHERE 股票代码 = '{stock_code}' 
            AND trade_date = '{trade_date}'
            LIMIT 1
        """
        df = pd.read_sql(sql, engine)
        if not df.empty:
            # 处理时间格式，可能是Timedelta或字符串
            zt_time = df.iloc[0]['首次涨停时间']
            if hasattr(zt_time, 'total_seconds'):
                # Timedelta类型
                total_seconds = int(zt_time.total_seconds())
                hours = total_seconds // 3600
                minutes = (total_seconds % 3600) // 60
                seconds = total_seconds % 60
                return f"{hours:02d}:{minutes:02d}:{seconds:02d}"
            else:
                return str(zt_time)
    except Exception as e:
        logger.warning(f"查询ztb_day失败: {e}")
    return None
```
