## 一、需求理解（更新）

### 用户最新思路
1. **时间轴刷新机制**:
   - 点击时间轴刷新按钮 → 刷新时间点列表
   - **默认不自动刷新**（与数据监控一致）
   - 只有刷新页面或点击刷新按钮时才刷新时间轴

2. **时间轴数据来源**:
   - **先查Redis**: 获取Redis中存储的时间戳列表
   - **Redis无数据**: 从`monitor_gp_sssj_{date}`表获取
   - **与数据监控完全一致**

3. **涨停股票筛选**:
   - 选择时间点后，查询该时间点的`monitor_gp_sssj`数据
   - 筛选`is_zt=1`的股票
   - 进行行概分析

### 关键变化
- **时间轴数据**: 从Redis/MySQL获取实际时间戳（非固定间隔）
- **刷新机制**: 手动刷新，非实时自动刷新
- **数据一致性**: 与数据监控页面完全一致

## 三、方案设计 V3

### 3.1 核心流程

```
用户打开页面
    ↓
加载时间轴（手动/页面刷新）
    1. 调用 /api/ztb/timestamps
    2. 先查Redis: key="ztb:timestamps:{date}"
    3. Redis无: 查MySQL monitor_gp_sssj_{date} 获取DISTINCT time
    4. 时间戳列表写入Redis缓存
    ↓
时间轴显示可用时间点
    ↓
用户点击时间点（或拖动滑块）
    ↓
查询该时间点涨停股票
    1. 调用 /api/ztb/list-by-time?date={date}&time={time}
    2. 先查Redis: key="monitor_gp_sssj_{date}:{time}"
    3. Redis无: 查MySQL monitor_gp_sssj_{date} WHERE time={time}
    4. 筛选 is_zt=1 的股票
    ↓
行概分析
    ↓
展示结果
```

```python
@ztb_bp.route('/api/ztb/timestamps')
def get_ztb_timestamps():
    """
    获取涨停选股时间轴时间戳列表
    
    逻辑:
    1. 先查Redis缓存
    2. Redis未命中，查MySQL monitor_gp_sssj_{date}表
    3. 返回DISTINCT time列表
    """
    try:
        date = request.args.get('date')
        if not date:
            return jsonify({'code': 400, 'message': '缺少date参数'}), 400
        
        date_str = date.replace('-', '')
        
        # 1. 先查Redis
        redis_key = f"ztb:timestamps:{date_str}"
        try:
            timestamps = redis_util.get(redis_key)
            if timestamps:
                logger.info(f"从Redis获取时间戳: {date}, 共{len(timestamps)}个")
                return jsonify({
                    'code': 0, 
                    'message': 'success',
                    'data': timestamps,
                    'source': 'redis'
                })
        except Exception as e:
            logger.warning(f"Redis查询失败: {e}")
        
        # 2. Redis未命中，查MySQL
        table_name = f"monitor_gp_sssj_{date_str}"
        sql = f"SELECT DISTINCT time FROM {table_name} ORDER BY time"
        
        try:
            df = pd.read_sql(sql, engine)
            timestamps = df['time'].tolist()
            logger.info(f"从MySQL获取时间戳: {date}, 共{len(timestamps)}个")
        except Exception as e:
            logger.error(f"MySQL查询失败: {e}")
            return jsonify({
                'code': 500, 
                'message': f'查询失败: {str(e)}'
            }), 500
        
        # 3. 写入Redis缓存（5分钟）
        try:
            redis_util.set(redis_key, timestamps, expire=300)
        except Exception as e:
            logger.warning(f"Redis写入失败: {e}")
        
        return jsonify({
            'code': 0,
            'message': 'success',
            'data': timestamps,
            'source': 'mysql'
        })
        
    except Exception as e:
        logger.error(f"获取时间戳失败: {e}")
        return jsonify({'code': 500, 'message': str(e)}), 500
```

### 4.2 服务层实现

```python
# ztb_analysis_service.py

def get_ztb_list_by_time_v3(date: str, time_str: str) -> Dict:
    """
    V3: 获取指定时间点的涨停股票列表
    
    与数据监控完全一致的数据查询逻辑:
    1. 先查Redis
    2. Redis未命中查MySQL
    3. 筛选is_zt=1
    
    Args:
        date: 日期，如 "2026-04-29"
        time_str: 时间，如 "10:00:00"
    
    Returns:
        {
            'zt_stocks': [...],      # 涨停股票列表
            'industries': [...],      # 行业分布
            'concepts': [...],        # 概念分布
            'stats': {...}            # 统计信息
        }
    """
    date_str = date.replace('-', '')
    redis_key = f"monitor_gp_sssj_{date_str}:{time_str}"
    
    df = None
    source = None
    
    # 1. 先查Redis（与数据监控完全一致）
    try:
        df = redis_util.load_dataframe_by_key(redis_key)
        if df is not None and not df.empty:
            source = 'redis'
            logger.info(f"从Redis获取数据: {redis_key}, {len(df)}条")
    except Exception as e:
        logger.warning(f"Redis查询失败: {e}")
    
    # 2. Redis未命中，查MySQL（与数据监控完全一致）
    if df is None or df.empty:
        try:
            table_name = f"monitor_gp_sssj_{date_str}"
            sql = f"""
                SELECT * FROM {table_name}
                WHERE time = %s
            """
            df = pd.read_sql(sql, engine, params=(time_str,))
            source = 'mysql'
            logger.info(f"从MySQL获取数据: {table_name}, time={time_str}, {len(df)}条")
        except Exception as e:
            logger.error(f"MySQL查询失败: {e}")
            raise
    
    # 3. 数据为空
    if df is None or df.empty:
        return {
            'zt_stocks': [],
            'industries': [],
            'concepts': [],
            'stats': {
                'total': 0,
                'source': None,
                'time': time_str,
                'date': date
            }
        }
    
    # 4. 筛选is_zt=1的股票
    if 'is_zt' not in df.columns:
        logger.error(f"数据缺少is_zt字段，可用字段: {df.columns.tolist()}")
        raise ValueError("数据缺少is_zt字段")
    
    zt_df = df[df['is_zt'] == 1].copy()
    total_zt = len(zt_df)
    
    logger.info(f"时间点{time_str}涨停股票: {total_zt}/{len(df)}")
    
    # 5. 处理股票列表
    zt_stocks = []
    for _, row in zt_df.iterrows():
        zt_stocks.append({
            'stock_code': str(row.get('stock_code', '')).zfill(6),
            'stock_name': row.get('short_name', ''),
            'price': float(row.get('price', 0)),
            'change_pct': float(row.get('change_pct', 0)),
            'main_net_amount': float(row.get('main_net_amount', 0)),
            'cumulative_main_net': float(row.get('cumulative_main_net', 0)),
            'time': time_str
        })
    
    # 6. 行概分析
    industries = analyze_industries_v3(zt_stocks)
    concepts = analyze_concepts_v3(zt_stocks)
    
    return {
        'zt_stocks': zt_stocks,
        'industries': industries,
        'concepts': concepts,
        'stats': {
            'total': total_zt,
            'source': source,
            'time': time_str,
            'date': date,
            'all_stocks': len(df)
        }
    }


def analyze_industries_v3(stocks: List[Dict]) -> List[Dict]:
    """分析行业分布V3"""
    if not stocks:
        return []
    
    # 获取行业信息
    stock_codes = [s['stock_code'] for s in stocks]
    industry_map = get_stock_industry_map(stock_codes)  # 从缓存或数据库
    
    # 统计
    industry_stats = {}
    for stock in stocks:
        code = stock['stock_code']
        industry = industry_map.get(code, '未知')
        
        if industry not in industry_stats:
            industry_stats[industry] = {
                'name': industry,
                'count': 0,
                'stocks': []
            }
        
        industry_stats[industry]['count'] += 1
        industry_stats[industry]['stocks'].append({
            'code': code,
            'name': stock['stock_name']
        })
    
    # 排序并返回
    result = sorted(
        industry_stats.values(),
        key=lambda x: x['count'],
        reverse=True
    )
    
    return result


def analyze_concepts_v3(stocks: List[Dict]) -> List[Dict]:
    """分析概念分布V3"""
    if not stocks:
        return []
    
    # 获取概念信息
    stock_codes = [s['stock_code'] for s in stocks]
    concept_map = get_stock_concept_map(stock_codes)  # 从缓存或数据库
    
    # 统计
    concept_stats = {}
    for stock in stocks:
        code = stock['stock_code']
        concepts = concept_map.get(code, [])
        
        for concept in concepts:
            if concept not in concept_stats:
                concept_stats[concept] = {
                    'name': concept,
                    'count': 0,
                    'stocks': []
                }
            
            concept_stats[concept]['count'] += 1
            concept_stats[concept]['stocks'].append({
                'code': code,
                'name': stock['stock_name']
            })
    
    # 排序并返回
    result = sorted(
        concept_stats.values(),
        key=lambda x: x['count'],
        reverse=True
    )
    
    return result[:20]  # 只返回前20个概念
```

### 3.3 服务层实现

```python
# ztb_analysis_service.py

def get_ztb_list_by_time(date: str, time_str: str) -> Dict:
    """
    获取指定时间点的涨停股票列表
    
    查询流程:
    1. 先查Redis
    2. Redis未命中，查MySQL
    3. 筛选is_zt=1的股票
    4. 进行行概分析
    
    Args:
        date: 日期，如 "2026-04-29"
        time_str: 时间，如 "10:00:00"
    
    Returns:
        {
            'zt_stocks': [...],        # 涨停股票列表
            'industry_analysis': {...}, # 行业分析
            'concept_analysis': {...},  # 概念分析
            'stats': {...}              # 统计信息
        }
    """
    # 1. 构建Redis key
    date_str = date.replace('-', '')
    redis_key = f"monitor_gp_sssj_{date_str}:{time_str}"
    
    df = None
    source = None
    
    # 2. 先查Redis
    try:
        df = redis_util.load_dataframe_by_key(redis_key)
        if df is not None and not df.empty:
            source = 'redis'
            logger.info(f"从Redis获取数据: {redis_key}, {len(df)}条")
    except Exception as e:
        logger.warning(f"Redis查询失败: {e}")
    
    # 3. Redis未命中，查MySQL
    if df is None or df.empty:
        try:
            table_name = f"monitor_gp_sssj_{date_str}"
            sql = f"""
                SELECT * FROM {table_name}
                WHERE time = %s
            """
            df = pd.read_sql(sql, engine, params=(time_str,))
            source = 'mysql'
            logger.info(f"从MySQL获取数据: {table_name}, time={time_str}, {len(df)}条")
        except Exception as e:
            logger.error(f"MySQL查询失败: {e}")
            raise
    
    if df is None or df.empty:
        return {
            'zt_stocks': [],
            'industry_analysis': {},
            'concept_analysis': {},
            'stats': {
                'total': 0,
                'source': None,
                'time': time_str
            }
        }
    
    # 4. 筛选is_zt=1的股票
    if 'is_zt' not in df.columns:
        logger.error(f"数据缺少is_zt字段，可用字段: {df.columns.tolist()}")
        raise ValueError("数据缺少is_zt字段")
    
    zt_df = df[df['is_zt'] == 1].copy()
    
    logger.info(f"时间点{time_str}涨停股票: {len(zt_df)}/{len(df)}")
    
    # 5. 基础字段处理
    zt_stocks = []
    for _, row in zt_df.iterrows():
        zt_stocks.append({
            'stock_code': str(row.get('stock_code', '')).zfill(6),
            'stock_name': row.get('short_name', ''),
            'price': float(row.get('price', 0)),
            'change_pct': float(row.get('change_pct', 0)),
            'main_net_amount': float(row.get('main_net_amount', 0)),
            'cumulative_main_net': float(row.get('cumulative_main_net', 0)),
            'time': time_str
        })
    
    # 6. 行概分析
    industry_analysis = analyze_industry_distribution(zt_stocks)
    concept_analysis = analyze_concept_distribution(zt_stocks)
    
    return {
        'zt_stocks': zt_stocks,
        'industry_analysis': industry_analysis,
        'concept_analysis': concept_analysis,
        'stats': {
            'total': len(zt_stocks),
            'source': source,
            'time': time_str,
            'date': date
        }
    }


def analyze_industry_distribution(stocks: List[Dict]) -> Dict:
    """分析涨停股票的行业分布"""
    if not stocks:
        return {}
    
    # 获取股票行业信息（从缓存或数据库）
    stock_codes = [s['stock_code'] for s in stocks]
    industry_map = get_stock_industries(stock_codes)
    
    # 统计行业分布
    industry_count = {}
    for stock in stocks:
        code = stock['stock_code']
        industry = industry_map.get(code, '未知')
        industry_count[industry] = industry_count.get(industry, 0) + 1
    
    # 排序
    sorted_industries = sorted(
        industry_count.items(), 
        key=lambda x: x[1], 
        reverse=True
    )
    
    return {
        'distribution': dict(sorted_industries),
        'top_industries': sorted_industries[:5]
    }


def analyze_concept_distribution(stocks: List[Dict]) -> Dict:
    """分析涨停股票的概念分布"""
    if not stocks:
        return {}
    
    # 获取股票概念信息
    stock_codes = [s['stock_code'] for s in stocks]
    concept_map = get_stock_concepts(stock_codes)
    
    # 统计概念分布
    concept_count = {}
    for stock in stocks:
        code = stock['stock_code']
        concepts = concept_map.get(code, [])
        for concept in concepts:
            concept_count[concept] = concept_count.get(concept, 0) + 1
    
    # 排序
    sorted_concepts = sorted(
        concept_count.items(),
        key=lambda x: x[1],
        reverse=True
    )
    
    return {
        'distribution': dict(sorted_concepts),
        'top_concepts': sorted_concepts[:10]
    }
```

```javascript
// 时间点转字符串
function pointToTime(point) {
    const [startHour, startMin] = TIMELINE_CONFIG.startTime.split(':').map(Number);
    const totalMinutes = startHour * 60 + startMin + point * TIMELINE_CONFIG.interval;
    
    // 处理午休
    if (totalMinutes >= 11 * 60 + 30 && totalMinutes < 13 * 60) {
        // 午休时间，跳到13:00
        return '13:00';
    }
    
    const hour = Math.floor(totalMinutes / 60);
    const min = totalMinutes % 60;
    return `${hour.toString().padStart(2, '0')}:${min.toString().padStart(2, '0')}`;
}

// 时间字符串转点
function timeToPoint(timeStr) {
    const [hour, min] = timeStr.split(':').map(Number);
    const [startHour, startMin] = TIMELINE_CONFIG.startTime.split(':').map(Number);
    
    let totalMinutes = hour * 60 + min;
    const startTotal = startHour * 60 + startMin;
    
    // 处理午休
    if (hour === 13 && min < 30) {
        totalMinutes = 11 * 60 + 30;  // 跳到午休前
    }
    
    return Math.floor((totalMinutes - startTotal) / TIMELINE_CONFIG.interval);
}
```

## 3. 涨停行概时间轴查询问题分析


## 问题描述

时间轴模式点击查询没有数据，但默认模式有数据。

## 对比分析

### 默认模式 (`get_ztb_tags`)

```python
def get_ztb_tags(date: str = None) -> dict:
    # 1. 获取涨停股票代码（当天）
    #    - 优先Redis: redis_util.get_zt_stocks_from_redis(date, 'stock')
    #    - 回退MySQL: SELECT DISTINCT stock_code FROM monitor_gp_sssj_{date} WHERE is_zt = 1
    
    # 2. 获取涨停股票代码（历史日期）
    #    - 从analysis_ztb_detail_{year}表查询
    
    # 3. 从宽表内存缓存获取行业和概念
    #    - 使用_stock_cache
    
    # 4. 统计并返回
```

**关键特点：**
- 使用 `redis_util.get_zt_stocks_from_redis(date, 'stock')` 获取当天涨停
- 历史日期使用 `analysis_ztb_detail_{year}` 表
- 查询条件是 `is_zt = 1`，没有时间限制（获取全天所有涨停）

### 时间轴模式 (`get_ztb_snapshot`)

```python
def get_ztb_snapshot(date: str, time: str) -> Dict:
    # 1. 获取指定时间点的涨停股票
    #    - Redis: monitor_gp_sssj_{date_str}:{time}
    #    - MySQL: SELECT stock_code FROM monitor_gp_sssj_{date} WHERE time = '{time}' AND is_zt = 1
    
    # 2. 分析行业和概念（复用逻辑）
```

**关键特点：**
- 必须指定时间点 `time`
- 查询特定时间点的涨停数据
- 如果该时间点没有涨停数据，返回空

## 问题原因

### 可能原因1：时间点没有涨停数据

`monitor_gp_sssj`表存储的是实时数据，每个时间点只存储当时的股票状态。

**问题：**
- 如果选择的 `time` 点没有涨停股票，`is_zt = 1` 的记录数为0
- 默认模式查询全天所有 `is_zt = 1` 的记录，不受时间限制

**示例：**
- 默认模式：查询 `2026-04-29` 全天所有涨停股票 → 返回100只
- 时间轴模式：查询 `2026-04-29 09:30:00` 的涨停股票 → 可能只有5只（开盘涨停）
- 时间轴模式：查询 `2026-04-29 09:30:15` 的涨停股票 → 可能有10只

### 可能原因2：Redis key格式不匹配

时间轴查询使用的Redis key：
```
monitor_gp_sssj_20260429:09:30:00
```

但实际存储的key可能不同（需要确认）。

### 可能原因3：时间格式不匹配

前端传递的时间格式与Redis/MySQL存储的格式不一致。

## 解决方案

### 方案A：沿用默认查询方式（推荐）

**核心思路：** 时间轴模式也应该查询全天涨停数据，而不是特定时间点。

**修改逻辑：**
```python
def get_ztb_snapshot(date: str, time: str) -> Dict:
    """获取指定时间点的涨停快照
    
    逻辑修改：
    - 不是查询该时间点的涨停股票
    - 而是查询到该时间点为止，所有曾经涨停的股票
    - 展示这些股票在该时间点的状态
    """
    date_str = date.replace('-', '')
    
    # 1. 获取到该时间点为止的所有涨停股票（沿用默认逻辑）
    #    - 使用ever_zt字段或查询历史涨停表
    
    # 2. 获取这些股票在该时间点的实时数据
    
    # 3. 分析行业和概念
```

**具体实施：**

1. **修改数据查询逻辑**
   - 查询 `monitor_gp_sssj_{date}` 表中 `ever_zt = 1` 且 `time <= '{time}'` 的最新记录
   - 或者查询 `analysis_ztb_detail_{year}` 表获取历史涨停

2. **修改 `_get_zt_stocks_at_time` 函数**
   ```python
   def _get_zt_stocks_at_time(date_str: str, time: str) -> List[str]:
       """获取到指定时间点为止的所有涨停股票"""

       # 或者查询历史涨停表
   ```

### 方案B：修复时间点查询

如果确实需要查询特定时间点的涨停，需要确认：
1. Redis中是否存在该时间点的数据
2. 时间格式是否正确
3. 该时间点是否真的有涨停股票

## 建议

**推荐方案A**，原因：
1. 与默认模式逻辑一致（查询涨停股票，不限制时间点）
2. 用户期望看到"到该时间点为止"的涨停情况

- 如果线程池任务失败，数据不会写入Redis

**可能原因3：Redis从未写入**
- 如果监控服务没有正常运行，Redis中不会有数据

当Redis中没有时间戳数据时，需要从MySQL获取。

```python
    获取涨停列表（支持时间筛选）
    
    Args:
        date: 日期
        time_str: 时间字符串，如 "10:00"，None表示全天
    
    Returns:
        涨停股票列表
    """
    base_query = f"""
        SELECT * FROM zt_records_{date}
        WHERE 1=1
    """
    
    if time_str:
        # 方案A：首次涨停时间在time_str之前（或等于）

        # 因为数据采集周期是3分钟
        base_query += f"""

        """
```

### 更精确的逻辑

考虑到数据采集周期是3分钟，时间轴应该显示：
- **该时间点新采集到的涨停股票**
- 即：`first_zt_time`在`(prev_time, current_time]`区间内的股票

```python
def get_ztb_list_by_time(date, current_time, prev_time=None):
    """
    获取指定时间点的涨停股票（新涨停）
    
    Args:
        date: 日期
        current_time: 当前时间，如 "10:00:00"
        prev_time: 上一时间，如 "09:57:00"，None表示从开盘开始
    
    Returns:
        在该时间窗口内首次涨停的股票
    """
    if prev_time is None:
        # 从开盘到current_time
        condition = f"first_zt_time <= '{current_time}'"
    else:
        # 在(prev_time, current_time]区间内首次涨停
        condition = f"first_zt_time > '{prev_time}' AND first_zt_time <= '{current_time}'"
    
    query = f"""
        SELECT * FROM zt_records_{date}
        WHERE {condition}
        ORDER BY first_zt_time
    """
    return execute_query(query)
```

### 服务层修改

```python
# ztb_analysis_service.py

def get_ztb_list(date, time_str=None, ...):
    """
    获取涨停列表
    
    Args:
        date: 日期
        time_str: 时间字符串，如 "10:00:00"
                None表示全天，不筛选时间
    """
    table_name = f"zt_records_{date.replace('-', '')}"
    
    base_sql = f"""
        SELECT * FROM {table_name}
        WHERE 1=1
    """
    
    if time_str:
        # 方案A：显示在该时间点新涨停的股票
        # 即first_zt_time在(prev_time, time_str]区间内
        prev_time = calculate_prev_time(time_str)  # 根据3分钟周期计算
        base_sql += f"""
            AND first_zt_time > '{prev_time}'
            AND first_zt_time <= '{time_str}'
        """
    
    # ... 其他筛选条件
    
    return execute_query(base_sql)

def calculate_prev_time(time_str, interval_minutes=3):
    """
    计算上一个时间点
    
    例如：
    - "10:00:00" -> "09:57:00" (3分钟间隔)
    - "09:30:00" -> "09:30:00" (开盘时间)
    """
    from datetime import datetime, timedelta
    
    t = datetime.strptime(time_str, '%H:%M:%S')
    
    # 如果早于09:30，返回09:30
    if t.hour < 9 or (t.hour == 9 and t.minute < 30):
        return '09:30:00'
    
    # 减去间隔
    prev = t - timedelta(minutes=interval_minutes)
    
    # 处理午休时间
    if prev.hour == 11 and prev.minute >= 30:
        # 11:30之后跳到13:00
        return '11:30:00'
    if prev.hour == 13 and prev.minute < 30:
        # 13:00-13:30之间，跳到11:30
        return '11:30:00'
    
    return prev.strftime('%H:%M:%S')
```

```javascript
function renderFlatTable(filteredGroups, container, isZtbPanel) {
    const sortState = isZtbPanel ? ztbSortState : crossSortState;
    
    // 1. 展平：将所有groups的stocks合并，注入match_count
    let allStocks = [];
    filteredGroups.forEach(group => {
        group.stocks.forEach(stock => {
            allStocks.push({ ...stock, match_count: group.match_count });
        });
    });

    // 2. 排序
    if (sortState.direction !== 'default') {
        const field = sortState.field;
        const asc = sortState.direction === 'asc';
        allStocks.sort((a, b) => {
            let va = parseFloat(a[field]) || 0;
            let vb = parseFloat(b[field]) || 0;
            return asc ? va - vb : vb - va;
        });
    } else {
        // 默认：match_count降序
        allStocks.sort((a, b) => (b.match_count || 0) - (a.match_count || 0));
    }

    // 3. 生成排序箭头
    function sortIcon(field) {
        if (sortState.field === field && sortState.direction !== 'default') {
            const arrow = sortState.direction === 'desc' ? '↓' : '↑';
            return `<span class="sort-arrow active">${arrow}</span>`;
        }
        return `<span class="sort-arrow">↕</span>`;
    }
    
    const panelFlag = isZtbPanel ? 'true' : 'false';

    // 4. 渲染单张表格
    container.innerHTML = `
        <table class="results-table">
            <thead><tr>
                <th>股票代码</th>
                <th>股票名称</th>
                <th class="sortable-header" onclick="handleFlatSort('change_pct', ${panelFlag})">
                    涨跌幅${sortIcon('change_pct')}</th>
                <th>现价</th>
                <th>债券代码</th>
                <th>债券名称</th>
                <th class="sortable-header" onclick="handleFlatSort('bond_change_pct', ${panelFlag})">
                    债券涨跌幅${sortIcon('bond_change_pct')}</th>
                <th class="sortable-header" onclick="handleFlatSort('match_count', ${panelFlag})">
                    命中个数${sortIcon('match_count')}</th>
                <th>所属行概</th>
            </tr></thead>
            <tbody>
                ${allStocks.map(stock => renderStockRow(stock)).join('')}
            </tbody>
        </table>
    `;
}
```

```javascript
function handleFlatSort(field, isZtbPanel) {
    const sortState = isZtbPanel ? ztbSortState : crossSortState;
    
    if (sortState.field === field) {
        // 同字段：desc → asc → default
        if (sortState.direction === 'desc') {
            sortState.direction = 'asc';
        } else if (sortState.direction === 'asc') {
            // 还原默认
            sortState.field = 'match_count';
            sortState.direction = 'desc';
        }
    } else {
        // 新字段，默认降序
        sortState.field = field;
        sortState.direction = 'desc';
    }
    
    // 重新渲染
    if (isZtbPanel) {
        renderResults(window.lastZtbQueryResult, 'ztbResultsSummary', 'ztbResultsContent');
    } else {
        renderResults(window.lastQueryResult);
    }
}
```

```javascript
    const changePct = parseFloat(stock.change_pct) || 0;
    const changeClass = changePct > 0 ? 'change-up' : (changePct < 0 ? 'change-down' : '');
    const changeSign = changePct > 0 ? '+' : '';

    const bondChangeClass = bondChangePct > 0 ? 'change-up' : (bondChangePct < 0 ? 'change-down' : '');

    const price = parseFloat(stock.price) || 0;
    
    // 所属行概标签（复用现有tags-grid逻辑）

        ...(stock.matched_industries || []).map(n => ({type:'industry', text:`🏭${n}`, fullText:n})),
        ...(stock.matched_concepts || []).map(n => ({type:'concept', text:`💡${n}`, fullText:n}))
    ];
    const showExpand = allTags.length > 5;
    const hiddenCount = allTags.length - 5;
    const tagsHtml = allTags.map((tag, i) => {
        const hidden = showExpand && i >= 5 ? ' hidden-tag' : '';
        return `<span class="tag-item tag-item-${tag.type}${hidden}" title="${tag.fullText}">${tag.text}</span>`;
    }).join('');
    const toggleBtn = showExpand ? 
        `<span class="expand-toggle-btn" onclick="toggleStockTags(this, '${stock.stock_code}')" title="展开/收起">+${hiddenCount}</span>` : '';
    
    // 命中个数列（无分组模式才显示，但函数中始终生成）
    const matchCountTd = stock.match_count !== undefined 
        ? `<td style="text-align:center; font-weight:bold; color:#2196F3;">${stock.match_count}</td>` : '';
    
    return `<tr>
        <td>${stock.stock_code}</td>
        <td>${stock.stock_name}</td>
        <td class="${changeClass}">${changeSign}${changePct.toFixed(2)}%</td>
        <td>${price > 0 ? price.toFixed(2) : '-'}</td>
        <td>${stock.bond_code || '-'}</td>
        <td>${stock.bond_name || '-'}</td>
        <td class="${bondChangeClass}">${bondChangeDisplay}</td>
        ${matchCountTd}
        <td><div class="tags-grid collapsed" id="tags-${stock.stock_code}">${tagsHtml}${toggleBtn}</div></td>
    </tr>`;
}
```

### 2.2 具体SQL分析

#### 问题SQL 1: 涨停股票查询
```sql
SELECT DISTINCT stock_code FROM monitor_gp_sssj_{date} WHERE is_zt = 1
```
**问题:**
- `is_zt` 字段无索引，全表扫描
- `DISTINCT` 需要排序去重

**优化:**
```sql
-- 添加索引
ALTER TABLE monitor_gp_sssj_{date} ADD INDEX idx_is_zt (is_zt);

-- 或使用覆盖索引
ALTER TABLE monitor_gp_sssj_{date} ADD INDEX idx_is_zt_code (is_zt, stock_code);
```

#### 问题SQL 2: 实时价格查询
```sql
SELECT stock_code, short_name, price, change_pct 
FROM monitor_gp_sssj_{date} 
WHERE time = (SELECT MAX(time) FROM monitor_gp_sssj_{date}) 
  AND stock_code IN ('000001', '000002', ...)
```
**问题:**
- 子查询 `(SELECT MAX(time) ...)` 每次执行
- `stock_code IN (...)` 列表可能很长（100+只）
- `time` 字段无索引

**优化:**
```sql
-- 先获取最大时间（缓存）
SET @max_time = (SELECT MAX(time) FROM monitor_gp_sssj_{date});

-- 再查询价格
SELECT stock_code, short_name, price, change_pct 
FROM monitor_gp_sssj_{date} 
WHERE time = @max_time AND stock_code IN (...);

-- 添加索引
ALTER TABLE monitor_gp_sssj_{date} ADD INDEX idx_time (time);
ALTER TABLE monitor_gp_sssj_{date} ADD INDEX idx_time_code (time, stock_code);
```

```python
def get_ztb_tags(date: str = None) -> dict:
    """获取涨停行业概念标签（带Redis缓存）"""
    if not date:
        date = datetime.now().strftime('%Y%m%d')
    
    # 1. 尝试从Redis获取
    redis_client = get_redis_client()
    cache_key_codes = f"ztb:codes:{date}"
    cache_key_industries = f"ztb:tags:{date}:industries"
    cache_key_concepts = f"ztb:tags:{date}:concepts"
    
    cached_codes = redis_client.smembers(cache_key_codes)
    if cached_codes:
        logger.info(f"从Redis获取涨停数据: {len(cached_codes)} 只")
        zt_codes = list(cached_codes)
        # 从Redis获取标签统计
        industries = redis_client.zrevrange(cache_key_industries, 0, -1, withscores=True)
        concepts = redis_client.zrevrange(cache_key_concepts, 0, -1, withscores=True)
        return {
            'date': date,
            'total_zt': len(zt_codes),
            'industries': [{'name': name, 'type': 'industry', 'code': get_code(name), 'count': int(count)} 
                          for name, count in industries],
            'concepts': [{'name': name, 'type': 'concept', 'code': get_code(name), 'count': int(count)} 
                        for name, count in concepts]
        }
    
    # 2. 从MySQL查询（原逻辑）
    ...
    
    # 3. 写入Redis缓存（设置过期时间5分钟）
    pipe = redis_client.pipeline()
    pipe.sadd(cache_key_codes, *zt_codes)
    pipe.expire(cache_key_codes, 300)
    
    for ind in industries:
        pipe.zadd(cache_key_industries, {ind['name']: ind['count']})
    pipe.expire(cache_key_industries, 300)
    
    for con in concepts:
        pipe.zadd(cache_key_concepts, {con['name']: con['count']})
    pipe.expire(cache_key_concepts, 300)
    
    pipe.execute()
    
    return result
```

**预期效果:**
- 首次查询: 与MySQL查询相当
- 后续查询: 毫秒级（Redis内存操作）

### 方案D: 缓存时间戳（避免重复子查询）

```python
# 全局缓存
_max_time_cache: Dict[str, str] = {}

def get_max_time(date: str, table_type: str = 'stock') -> str:
    """获取最大时间戳（带缓存）"""
    cache_key = f"{table_type}:{date}"
    
    # 检查缓存（5秒过期）
    if cache_key in _max_time_cache:
        cached_time, cached_at = _max_time_cache[cache_key]
        if time.time() - cached_at < 5:  # 5秒内有效
            return cached_time
    
    # 查询数据库
    mysql_tool = mysql_util.get_mysql_tool()
    table = f"monitor_gp_sssj_{date}" if table_type == 'stock' else f"monitor_zq_sssj_{date}"
    
    with mysql_tool.engine.connect() as conn:
        result = conn.execute(text(f"SELECT MAX(time) as max_time FROM {table}")).fetchone()
        max_time = result[0] if result else None
    
    # 写入缓存
    if max_time:
        _max_time_cache[cache_key] = (max_time, time.time())
    
    return max_time
```

### 4.1 服务层新增函数 (`stock_picker_service.py`)

```python
def get_ztb_tags(date: str = None) -> dict:
    """
    获取涨停板的行业和概念标签
    
    数据源策略：
    - 当天日期：从 monitor_gp_sssj_{date} 表查 is_zt=1
    - 历史日期：从 analysis_ztb_detail_{year} 表查 trade_date
    
    然后通过股票代码查宽表获取完整行业和概念
    """
    from datetime import datetime
    
    if not date:
        date = datetime.now().strftime('%Y%m%d')
    
    today = datetime.now().strftime('%Y%m%d')
    mysql_tool = mysql_util.get_mysql_tool()
    
    # 1. 获取涨停股票代码
    zt_codes = []
    
    if date == today:
        # 当天：从实时监控表查 is_zt=1
        try:
            with mysql_tool.engine.connect() as conn:
                sql = f"SELECT DISTINCT stock_code FROM monitor_gp_sssj_{date} WHERE is_zt = 1"
                rows = pd.read_sql(sql, conn).to_dict('records')
                zt_codes = [r['stock_code'] for r in rows]
        except Exception as e:
            logger.warning(f"实时监控表查询失败({date}): {e}")
    
    if not zt_codes:
        # 历史日期 或 当天实时表无数据：从涨停分析表查
        try:
            year = date[:4]
            table = f"analysis_ztb_detail_{year}"
            date_fmt = f"{date[:4]}-{date[4:6]}-{date[6:8]}"
            with mysql_tool.engine.connect() as conn:
                sql = f"SELECT DISTINCT stock_code FROM {table} WHERE trade_date = '{date_fmt}'"
                rows = pd.read_sql(sql, conn).to_dict('records')
                zt_codes = [r['stock_code'] for r in rows]
        except Exception as e:
            logger.error(f"涨停分析表查询失败({date}): {e}")
    
    if not zt_codes:
        return {'date': date, 'total_zt': 0, 'industries': [], 'concepts': []}
    
    # 2. 从宽表获取完整行业和概念
    # 确保内存缓存已加载
    if not _stock_cache:
        load_memory_cache()
    
    industry_counter = defaultdict(int)  # 行业名 → 出现次数
    concept_counter = defaultdict(int)   # 概念名 → 出现次数
    
    for code in zt_codes:
        stock_data = _stock_cache.get(code)
        if stock_data:
            for ind in stock_data['industries']:
                industry_counter[ind] += 1
            for con in stock_data['concepts']:
                concept_counter[con] += 1
    
    # 3. 按频次降序排列
    industries = sorted(
        [{'name': name, 'type': 'industry', 'count': count} 
         for name, count in industry_counter.items()],
        key=lambda x: x['count'], reverse=True
    )
    
    concepts = sorted(
        [{'name': name, 'type': 'concept', 'count': count} 
         for name, count in concept_counter.items()],
        key=lambda x: x['count'], reverse=True
    )
    
    return {
        'date': date,
        'total_zt': len(zt_codes),
        'industries': industries,
        'concepts': concepts
    }
```
