```python
def get_effective_trade_date(target_date: str = None) -> str:
    """
    获取有效的交易日期
    - 如果target_date是交易日，返回target_date
    - 如果不是交易日，返回上一个交易日
    
    Args:
        target_date: 目标日期(YYYYMMDD)，None则使用当天
        
    Returns:
        有效的交易日期(YYYYMMDD)
    """
    if target_date is None:
        target_date = time.strftime("%Y%m%d", time.localtime())
    
    # 查询data_jyrl表判断是否是交易日
    sql = """
        SELECT trade_date 
        FROM data_jyrl 
        WHERE trade_date <= :target_date AND trade_status = '1'
        ORDER BY trade_date DESC 
        LIMIT 1
    """
    
    with engine.connect() as conn:
        result = conn.execute(text(sql), {'target_date': target_date})
        row = result.fetchone()
        
        if row:
            return row[0]
        else:
            # 如果没有找到，返回target_date（兜底）
            return target_date
```

```python
def _create_stock_spot_em_indexes():
    """
    为data_stock_spot_em表创建索引（如果不存在）
    """
    table_name = "data_stock_spot_em"
    
    indexes = [
        ("idx_trade_date", "trade_date"),
        ("idx_stock_code", "stock_code"),
        ("idx_circulating_cap", "circulating_market_cap"),
        ("idx_trade_date_code", "trade_date, stock_code"),
    ]
    
    with engine.connect() as conn:
        for idx_name, idx_columns in indexes:
            try:
                # 检查索引是否存在
                check_sql = f"""
                    SELECT COUNT(*) FROM information_schema.STATISTICS 
                    WHERE TABLE_SCHEMA = DATABASE() 
                    AND TABLE_NAME = '{table_name}' 
                    AND INDEX_NAME = '{idx_name}'
                """
                result = conn.execute(text(check_sql))
                count = result.scalar()
                
                if count == 0:
                    # 创建索引
                    create_sql = f"CREATE INDEX `{idx_name}` ON `{table_name}` ({idx_columns})"
                    conn.execute(text(create_sql))
                    logger.info(f"[stock_spot_em] 创建索引：{idx_name} ({idx_columns})")
            except Exception as e:
                logger.warning(f"[stock_spot_em] 创建索引 {idx_name} 失败：{e}")
```

## 四、前端实时计算函数

```javascript
/**
 * 获取龙头展示role（前端实时计算）
 * 
 * @param {Object} stock - 股票对象
 * @param {Array} mainlineStocks - 同主线所有股票
 * @param {String} tradeDate - 交易日期YYYYMMDD
 * @returns {Object} - {displayRole, leaderType, marketCap}
 */
async function getLeaderDisplayRole(stock, mainlineStocks, tradeDate) {
    const result = {
        originalRole: stock.role,
        displayRole: stock.role,
        leaderType: null,  // 权重龙头/板块中军/小盘情绪龙
        marketCap: null,
        isEstimated: false
    };
    
    // 1. 如果AI已定义细分类型，直接返回
    if (['权重龙头', '板块中军', '小盘情绪龙'].includes(stock.role)) {
        result.leaderType = stock.role;
        return result;
    }
    
    // 2. 如果AI只定义了'龙头'，需要根据市值细分
    if (stock.role === '龙头') {
        // 查询流通市值
        const marketCap = await queryMarketCap(stock.stock_code, tradeDate);
        result.marketCap = marketCap;
        
        if (marketCap) {
            if (marketCap > 500) {
                result.leaderType = '权重龙头';
            } else if (marketCap >= 200) {
                result.leaderType = '板块中军';
            } else {
                result.leaderType = '小盘情绪龙';
            }
            result.displayRole = result.leaderType + '(细分)';
            result.isEstimated = true;
        } else {
            // 无市值数据，保持原样
            result.leaderType = '龙头';
        }
    }
    
    // 3. 如果AI未定义龙头，检查是否需要按时间补龙头
    // （这部分逻辑根据您的需求决定是否启用）
    const hasAiLeader = mainlineStocks.some(s => 
        ['龙头', '权重龙头', '板块中军', '小盘情绪龙'].includes(s.role)
    );
    
    if (!hasAiLeader && stock.role !== '龙头') {
        // AI未定义龙头，按时间排序看是否是第1只
        const sortedStocks = mainlineStocks.sort((a, b) => 
            a.anomaly_time.localeCompare(b.anomaly_time)
        );
        const rank = sortedStocks.findIndex(s => s.stock_code === stock.stock_code);
        
        if (rank === 0) {
            // 第1只，查询市值判断类型
            const marketCap = await queryMarketCap(stock.stock_code, tradeDate);
            result.marketCap = marketCap;
            
            if (marketCap) {
                if (marketCap > 500) {
                    result.leaderType = '权重龙头';
                } else if (marketCap >= 200) {
                    result.leaderType = '板块中军';
                } else {
                    result.leaderType = '小盘情绪龙';
                }
                result.displayRole = result.leaderType + '(估算)';
                result.isEstimated = true;
            }
        }
    }
    
    return result;
}

/**
 * 查询流通市值
 */
async function queryMarketCap(stockCode, tradeDate) {
    try {
        const response = await fetch(`/api/stock/market-cap?code=${stockCode}&date=${tradeDate}`);
        const data = await response.json();
        return data.circulating_market_cap;
    } catch (e) {
        console.error('查询流通市值失败:', e);
        return null;
    }
}
```

```python
def classify_leader_by_market_cap(stock_code: str, trade_date: str) -> str:
    """
    根据流通市值判断龙头细分类型
    
    Args:
        stock_code: 股票代码
        trade_date: 交易日期YYYYMMDD
        
    Returns:
        '权重龙头' / '板块中军' / '小盘情绪龙' / '未知'
    """
    # 查询该股票当天的流通市值
    sql = """
        SELECT circulating_market_cap 
        FROM data_stock_spot_em 
        WHERE stock_code = :stock_code AND trade_date = :trade_date
        LIMIT 1
    """
    
    with engine.connect() as conn:
        result = conn.execute(text(sql), {
            'stock_code': stock_code,
            'trade_date': trade_date
        })
        row = result.fetchone()
        
        if not row or row[0] is None:
            return '未知'
        
        market_cap = float(row[0])
        
        # 根据市值判断类型
        if market_cap > 500:
            return '权重龙头'
        elif 200 <= market_cap <= 500:
            return '板块中军'
        else:
            return '小盘情绪龙'
```

```javascript
// 前端展示逻辑
function getDisplayRole(stock, mainlineStocks) {
    // 1. 检查AI是否已定义龙头
    const hasAiLeader = mainlineStocks.some(s => 
        s.role === '权重龙头' || s.role === '板块中军' || 
        s.role === '小盘情绪龙' || s.role === '龙头'
    );
    
    if (hasAiLeader) {
        // AI已定义龙头，直接显示数据库中的role
        return stock.role;
    }
    
    // 2. AI未定义龙头，按简化版规则计算展示role
    const sortedStocks = mainlineStocks.sort((a, b) => 
        a.anomaly_time.localeCompare(b.anomaly_time)
    );
    const rank = sortedStocks.findIndex(s => s.stock_code === stock.stock_code);
    
    if (rank === 0) {
        // 第1只：根据市值判断展示role
        const marketCap = stock.circulating_market_cap || 0;
        if (marketCap > 500) return '权重龙头(估算)';
```

```python
    """
    按时间排序，根据市值细分龙头类型
    """
    sorted_stocks = sorted(stocks, key=lambda x: x['anomaly_time'])

        first_stock['new_role'] = '权重龙头'
        first_stock['score'] = 100
    elif 200 <= market_cap <= 500:
        first_stock['new_role'] = '板块中军'
        first_stock['score'] = 100
    else:
        first_stock['new_role'] = '小盘情绪龙'
        first_stock['score'] = 100

    for i in [1, 2]:
```

```python
def keep_ai_leader_and_recalc_others(engine, mainline_id, stocks):
    """
    保留AI定义的龙头，对其他股票重新计算role
    """
    # 找出AI定义的龙头
    ai_leader = [s for s in stocks if s['role'] in ['权重龙头', '板块中军', '小盘情绪龙', '龙头']]
    
    # 对其他股票按时间排序
    other_stocks = [s for s in stocks if s not in ai_leader]
    other_stocks.sort(key=lambda x: x['anomaly_time'])
    
    # 对其他股票重新分配role（不再有龙头，只有补涨和跟风）
    for i, stock in enumerate(other_stocks):
        if i < 2:  # 第1-2只其他股票 → 补涨
            stock['new_role'] = '补涨'
            stock['score'] = max(70 - i * 15, 20)
        else:  # 第3只及以后 → 跟风
            stock['new_role'] = '跟风'
            stock['score'] = max(40 - (i - 2) * 5, 20)
    
    return stocks
```

```python
class LeaderAnalyzerV2:
    """龙头分析器V2（同主线横向比较）"""
    
    def analyze_mainline(self, mainline_name: str, date: str):
        """分析单个主线的龙头"""
        # 1. 获取该主线所有涨停股票
        stocks = self.get_stocks_by_mainline(mainline_name, date)
        
        if len(stocks) < 2:
            logger.info(f"主线 {mainline_name} 只有 {len(stocks)} 只股票，无需比较")
            return
        
        # 2. 构建横向比较Prompt
        prompt = self.build_comparison_prompt(mainline_name, stocks)
        
        # 3. AI分析
        response = self.call_ai(prompt)
        
        # 4. 解析并更新
        analysis = json.loads(response)
        for stock in analysis:
            self.update_stock_anomaly(stock)
    
    def analyze_all_mainlines(self, date: str):
        """分析当日所有主线的龙头"""
        mainlines = self.get_mainlines_by_date(date)
        
        for mainline in mainlines:
            if mainline['stock_count'] >= 3:  # 至少3只才分析龙头
                self.analyze_mainline(mainline['mainline_name'], date)
```

```python
            print(f"  {table}: {count}条")
        
        if total_old == 0:

            return False

        return True
```


**问题**：
- 表数量爆炸
- 查询性能差（多表JOIN）

- 扩展困难（每加一个维度建新表）

## 8. 龙头股体量维度细分方案-基于游资实战体系


## 方案概述

**设计目标**：基于体量维度将龙头细分为三类，便于盘中快速识别和策略匹配

**适用场景**：`stock_anomaly` 盘中异动实时分析

**核心维度**：流通市值（最直观、最可量化）

---

## 一、三类龙头定义

### 1. 权重龙头（指数大盘型）

**定义**：
- 流通市值 > 500亿
- 对指数有显著影响（沪深300/中证500成分股）
- 机构资金主导，趋势性强

**特征**：
| 维度 | 特征 |
|------|------|
| 涨停时间 | 通常较晚（10:00后），需要大量资金封板 |
| 封单特点 | 封单金额巨大（>10亿），但封单比可能不高 |
| 连板能力 | 较弱，通常1-2板，以趋势为主 |
| 带动效应 | 带动指数和板块情绪，但非短线标杆 |
| 参与资金 | 机构、大游资、量化资金 |
| 操作策略 | 趋势跟随，不追高，回调低吸 |

**案例**：
- 宁德时代（新能源权重龙头）
- 贵州茅台（消费权重龙头）
- 比亚迪（汽车权重龙头）
- 中芯国际（半导体权重龙头）

**识别标准**：
```python
if 流通市值 > 500亿:
    类型 = "权重龙头"
    策略 = "趋势跟随，不追高"
```

---

### 2. 板块中军（容量中坚型）

**定义**：
- 流通市值 200-500亿
- 板块内市值最大或前列
- 机构与游资共同参与

**特征**：
| 维度 | 特征 |
|------|------|
| 涨停时间 | 中等（09:30-10:00），需要一定资金但可封住 |
| 封单特点 | 封单金额适中（1-5亿），封单比中等 |
| 连板能力 | 中等，通常2-3板，可趋势可短线 |
| 带动效应 | 板块核心标杆，后排跟风明显 |
| 参与资金 | 机构+游资合力 |
| 操作策略 | 可打板可低吸，板块核心 |

**案例**：
- 科大讯飞（AI中军，300亿+）
- 隆基绿能（光伏中军，曾300亿+）
- 海康威视（AI视觉中军）
- 北方华创（半导体设备中军）

**识别标准**：
```python
if 200亿 <= 流通市值 <= 500亿:
    类型 = "板块中军"
    策略 = "板块核心，可打板可低吸"
```

---

### 3. 小盘情绪龙（短线连板型）

**定义**：
- 流通市值 < 200亿
-  preferably 50-150亿最佳
- 游资主导，情绪标杆

**特征**：
| 维度 | 特征 |
|------|------|

```python
def classify_leader_by_size(stock: dict) -> dict:
    """
    基于体量维度分类龙头
    
    Args:
        stock: 包含市值、时间、连板数的股票数据
    
    Returns:
        分类结果
    """
    market_cap = stock['float_market_cap']  # 流通市值（亿）
    
    # 1. 按市值分类
    if market_cap > 500:
        size_type = "权重龙头"
        strategy = "趋势跟随，不追高，回调低吸"
        risk_level = "低"
        return_expect = "10-30%"
        fit_for = "稳健型"
    elif 200 <= market_cap <= 500:
        size_type = "板块中军"
        strategy = "板块核心，可打板可低吸"
        risk_level = "中"
        return_expect = "30-50%"
        fit_for = "平衡型"
    else:  # < 200亿
        size_type = "小盘情绪龙"
        strategy = "打板为主，快进快出"
        risk_level = "高"
        return_expect = "50-100%+"
        fit_for = "激进型"
    
    # 2. 细化特征
    time_feature = classify_time_feature(stock['anomaly_time'])
    seal_expect = classify_seal_expect(size_type)
    continuous_expect = classify_continuous_expect(stock['continuous_zt'], size_type)
    capital_type = classify_capital_type(size_type)
    
    return {
        'size_type': size_type,
        'strategy': strategy,
        'risk_level': risk_level,
        'return_expect': return_expect,
        'fit_for': fit_for,
        'time_feature': time_feature,
        'seal_expect': seal_expect,
        'continuous_expect': continuous_expect,
        'capital_type': capital_type
    }


def classify_time_feature(anomaly_time) -> str:
    """分类涨停时间特征"""
    time_str = str(anomaly_time)
    if '09:25' in time_str or '09:30' in time_str:
        return "早"
    elif '09:3' in time_str or '09:4' in time_str or '09:5' in time_str or '10:' in time_str:
        return "中"
    else:
        return "晚"


def classify_seal_expect(size_type: str) -> str:
    """分类封单预期"""
    mapping = {
        "权重龙头": "大",
        "板块中军": "中",
        "小盘情绪龙": "小"
    }
    return mapping.get(size_type, "中")


def classify_continuous_expect(continuous_zt: int, size_type: str) -> str:
    """分类连板预期"""
    if size_type == "权重龙头":
        return "1-2板"
    elif size_type == "板块中军":
        return "2-3板"
    else:  # 小盘情绪龙
        if continuous_zt >= 2:
            return "5板+"
        else:
            return "2-3板"


def classify_capital_type(size_type: str) -> str:
    """分类资金类型"""
    mapping = {
        "权重龙头": "机构主导",
        "板块中军": "机构游资合力",
        "小盘情绪龙": "游资主导"
    }
    return mapping.get(size_type, "未知")
```

## 用户核心疑问分析

### 疑问1：主线纯度作为评分标准的问题

**用户观点**：
> 如果某股票与主线关联度最强，但它有多条主线，岂不是算不上龙头？

**问题本质**：
- 当前方案：主线越少纯度越高 → 可能排除真正龙头
- 真实情况：龙头往往是**多主线共振**的结果

**案例反驳**：

| 股票 | 主线 | 真实情况 |
|------|------|---------|
| 宁德时代 | 锂电池+新能源车+储能+创业板权重 | 新能源龙头，多主线共振 |
| 贵州茅台 | 白酒+消费+核心资产+MSCI | 消费龙头，多主线共振 |
| 中芯国际 | 半导体+国产替代+科创板+大基金 | 半导体龙头，多主线共振 |

**结论**：真正的龙头往往是**多主线交叉的核心标的**，而非单主线独苗。

### 疑问2：价格强度作为评分标准的问题

**用户观点**：
> 龙头主要谈的不是正宗不正宗吗？价格为什么重要？

**问题本质**：
- 当前方案：价格高=强度大 → 逻辑错误
- 真实情况：龙头看**逻辑正宗性**，而非价格绝对值

**案例反驳**：

| 股票 | 价格 | 主线地位 | 真实角色 |
|------|------|---------|---------|

**结论**：价格与龙头地位无直接关系，**逻辑正宗性**才是关键。

## 顶级游资历史经验总结

### 经验1：龙头识别的核心维度（来自一线游资）

```
真龙头 = 逻辑最正宗 + 资金最认可 + 带动效应最强

逻辑正宗性（40%权重）：
  - 主营业务与主线直接相关（非沾边）
  - 行业地位领先（市占率、技术壁垒）
  - 业绩可验证（非纯概念）

资金认可度（30%权重）：
  - 涨停时间最早（情绪拐点）
  - 封单坚决（不反复开板）
  - 连板能力强（资金持续接力）

带动效应（30%权重）：
  - 涨停后同主线个股跟涨
  - 板块情绪被激活
  - 后排补涨出现
```

### 经验2：多主线龙头的特殊性（来自量化研究）

```
多主线共振 = 加分项，而非减分项

逻辑：
  - 单主线：只有一条逻辑支撑，风险集中
  - 多主线：多重逻辑交叉，抗风险能力强
  - 真正的龙头往往是"多主线交叉的核心"

案例：
  2023年ChatGPT行情：
  - 单主线：某纯AI小票 → 涨幅有限
  - 多主线：科大讯飞（AI+教育+大模型+业绩）→ 真龙头
```

### 经验3：价格与龙头地位无关（来自打板族经验）

```
价格高低 ≠ 龙头地位

关键指标：
  - 流通市值：50-200亿最佳（太大拉不动，太小易控盘）
  - 股性活跃度：近期涨停次数、连板历史
  - 筹码结构：套牢盘少、获利盘健康

反例：
  - 高价股（>100元）可能流动性差
  - 低价股（<5元）可能基本面有问题
```

   - 核心业务直接受益 + 行业龙头 + 业绩可验证 = 35分
   - 主营业务相关 + 行业前列 = 25-30分
   - 边缘业务沾边 = 10-15分

2. 资金认可度（权重30%）
   【判断标准】
   - 涨停时间：越早越好（情绪拐点）
   - 涨停顺序：主线内第1只 > 第2-3只 > 第4只以后
   - 连板基因：continuous_zt越高，资金认可度越强
   
   【评分】
   - 主线内第1只涨停 + 2连板以上 = 30分
   - 主线内前3只涨停 + 首板 = 20-25分
   - 主线内第4只以后 = 10-15分

3. 带动效应（权重25%）- 龙头必备
   【判断标准】
   - 该股票涨停后，同主线其他股票是否跟涨
   - 是否激活了板块情绪（后排补涨出现）
   - 是否成为板块情绪标杆
   
   【说明】
   - 最早涨停的股票天然具有带动效应
   - 如果连板，带动效应更强
   - 跟风股无带动效应，只是被动上涨
   
   【评分】
   - 明显带动板块 + 后排补涨 = 25分
   - 有一定带动 = 15-20分
   - 无明显带动 = 5-10分

4. 市值适中性（权重10%）
   【判断标准】
   - 流通市值50-200亿：最佳（流动性好，拉升难度适中）
   - 流通市值<30亿：太小（易被控盘，持续性差）
   - 流通市值>500亿：太大（拉升难度大，难成短线龙头）
   
   【评分】
   - 50-200亿 = 10分
   - 30-50亿或200-500亿 = 5-8分
   - <30亿或>500亿 = 0-3分

【主线数量的正确理解】

⚠️ 重要：主线数量不是越少越好！

- 单主线：只有一条逻辑，风险集中，可能是独苗而非龙头
- 双主线：两条逻辑交叉，有一定支撑
- 三主线及以上：多主线共振，逻辑最硬，往往是真龙头

```python
def prepare_mainline_stocks(date: str, min_stocks: int = 3) -> list:
    """
    按主线分组准备涨停股票数据（基于stock_anomaly表）
    
    Args:
        date: 交易日期，如 '2026-06-29'
        min_stocks: 主线最少股票数（少于这个数不分析龙头）
    
    Returns:
        [{mainline_name, stocks: [...]}, ...]
    """
    engine = create_engine(config_util.get_config('common.url'))
    
    sql = f"""
    SELECT 
        stock_code,
        stock_name,
        anomaly_time,
        price,
        change_pct,
        continuous_zt,
        mainline_names,
        related_industries,
        related_concepts
    FROM stock_anomaly 
    WHERE trading_date = '{date}'
      AND anomaly_type = 'zt_hit'
    ORDER BY anomaly_time
    """
    
    with engine.connect() as conn:
        df = pd.read_sql(sql, conn)
    
    # 解析mainline_names JSON，按主线分组
    mainlines_dict = {}
    
    for _, row in df.iterrows():
        mainlines = json.loads(row['mainline_names']) if row['mainline_names'] else []
        
        for mainline in mainlines:
            if mainline not in mainlines_dict:
                mainlines_dict[mainline] = []
            mainlines_dict[mainline].append(row.to_dict())
    
    # 筛选股票数>=min_stocks的主线
    result = []
    for mainline, stocks in mainlines_dict.items():
        if len(stocks) >= min_stocks:
            # 按时间排序
            stocks_sorted = sorted(stocks, key=lambda x: str(x['anomaly_time']))
            result.append({
                'mainline_name': mainline,
                'stocks': stocks_sorted,
                'count': len(stocks_sorted)
            })
    
    # 按股票数降序排序（主线越强越优先分析）
    result.sort(key=lambda x: x['count'], reverse=True)
    
    return result


def format_stocks_for_prompt(stocks: list) -> str:
    """格式化股票列表为Prompt文本"""
    lines = []
    for i, s in enumerate(stocks, 1):
        mainlines = json.loads(s['mainline_names']) if s['mainline_names'] else []
        mainlines_str = ', '.join(mainlines[:3])  # 最多显示3个主线
        
        lines.append(f"""
{i}. 股票代码：{s['stock_code']}  股票名称：{s['stock_name']}
   涨停时间：{s['anomaly_time']}
   当前连板数：{s['continuous_zt']}板
   涨停价格：{s['price']}元
   涨跌幅：{s['change_pct']}%
   关联主线：[{mainlines_str}]{'...' if len(mainlines) > 3 else ''}
""")
```

```python
    """

            mainline_name=mainline['mainline_name'],
            stock_count=mainline['count'],
            stocks_list=format_stocks_for_prompt(mainline['stocks'])
        )
        
        # 调用AI
        try:
            if ai_engine == 'volcengine':
                from gs2026.analysis.worker.message.huoshanfangzhou.volcengine_client import volcengine_analysis
                response = volcengine_analysis(prompt)
            else:
                from gs2026.analysis.worker.message.deepseek.deepseek_analysis_event_driven import deepseek_analysis
                response = deepseek_analysis(prompt, _headless=True)
            
            # 解析结果
            result = json.loads(response)
            
            results.append({
                'mainline_name': mainline['mainline_name'],
                'stock_count': mainline['count'],
                'analysis': result,  # AI返回的数组
                'raw_response': response

        except Exception as e:
            logger.error(f"[龙头股识别] {mainline['mainline_name']} 分析失败: {e}")
```

```python
def prepare_block_stocks(date: str, min_stocks: int = 3) -> list:
    """
    按板块分组准备涨停股票数据
    
    Args:
        date: 交易日期，如 '20260629'
        min_stocks: 板块最少股票数（少于这个数不分析龙头）
    
    Returns:
        [{block_name, stocks: [...]}, ...]
    """
    engine = create_engine(config_util.get_config('common.url'))
    
    sql = f"""
    SELECT 
        股票代码,
        股票简称,
        首次涨停时间,
        涨停封单额,
        涨停封流比,
        连续涨停天数,
        涨停开板次数,
        a股流通市值,
        涨停原因类别,
        几天几板
    FROM ztb_day 
    WHERE trade_date = '{date}'
      AND 涨停 = '涨停'
    ORDER BY 首次涨停_time
    """
    
    with engine.connect() as conn:
        df = pd.read_sql(sql, conn)
    
    # 按板块分组
    blocks = []
    for block_name, group in df.groupby('涨停原因类别'):
        if len(group) >= min_stocks:  # 只分析有3只以上涨停的板块
            stocks = group.to_dict('records')
            blocks.append({
                'block_name': block_name,
                'stocks': stocks,
                'count': len(stocks)
            })
    
    return blocks


def format_stocks_list(stocks: list) -> str:
    """格式化股票列表为Prompt文本"""
    lines = []
    for i, s in enumerate(stocks, 1):
        lines.append(f"""
{i}. 股票代码：{s['股票代码']}  股票简称：{s['股票简称']}
   首次涨停时间：{s['首次涨停时间']}
   涨停封单额：{s['涨停封单额']}万
   涨停封流比：{s['涨停封流比']}%
   连续涨停天数：{s['连续涨停天数']}天
   涨停开板次数：{s['涨停开板次数']}次
   a股流通市值：{s['a股流通市值']}亿
   几天几板：{s['几天几板']}
""")
    return '\n'.join(lines)
```

```python
def analyze_leaders_by_block(date: str, ai_engine: str = 'volcengine') -> list:
    """
    按板块批量分析龙头股
    
    Args:
        date: 交易日期
        ai_engine: AI引擎 'volcengine' 或 'deepseek'
    
    Returns:
        分析结果列表
    """
    blocks = prepare_block_stocks(date)
    results = []
    
    for block in blocks:
        # 构建Prompt
        prompt = ENHANCED_ZTB_PROMPT.format(
            block_name=block['block_name'],
            block_count=block['count'],
            stocks_list=format_stocks_list(block['stocks'])
        )
        
        # 调用AI
        if ai_engine == 'volcengine':
            from gs2026.analysis.worker.message.huoshanfangzhou.volcengine_client import volcengine_analysis
            response = volcengine_analysis(prompt)
        else:
            from gs2026.analysis.worker.message.deepseek.deepseek_analysis_event_driven import deepseek_analysis
            response = deepseek_analysis(prompt, _headless=True)
        
        # 解析结果
        try:
            result = json.loads(response)
            results.append({
                'block_name': block['block_name'],
                'analysis': result,
                'raw_response': response
            })
        except json.JSONDecodeError:
            logger.error(f"解析失败: {block['block_name']}")
            continue
    
    return results
```

## 12. 龙头股识别算法改进方案


## 当前问题

**现象**：32只首次涨停的AI应用与智能体股票，AI分析没有识别出龙头股

**根本原因**：
1. 当前prompt只要求分析单只股票的涨停原因，没有横向比较
2. 缺乏"板块内地位"判断维度
3. 没有涨停时间序列分析（谁先涨停谁后涨停）
4. 缺乏封单强度、换手率等盘口数据

---

## 龙头股识别的真实逻辑（历史经验）

### 1. 时间优先原则
```
同板块内，先涨停的是龙头，后涨停的是跟风
- 10:00:00 涨停 → 龙头候选
- 10:05:00 涨停 → 跟风（受龙头带动）
- 10:30:00 涨停 → 补涨（板块已确认强势）
```

### 2. 封单强度原则
```
封单金额 / 流通市值 比率越高，龙头地位越稳
- 封单比 > 10% → 绝对龙头
- 封单比 5-10% → 强势龙头
- 封单比 < 5% → 弱势或跟风
```

### 3. 连板基因原则
```
历史连板能力决定龙头气质：
- 近30日有过连板 → 龙头基因
- 股性活跃（常涨停）→ 受资金青睐
- 首次涨停且历史活跃 → 潜力龙头
```

### 4. 板块贡献度原则
```
对板块情绪带动的贡献：
- 涨停后板块内其他股票跟风上涨 → 龙头确认
- 涨停时板块尚无其他涨停 → 开拓者（高风险高回报）
- 涨停时板块已有多个涨停 → 跟风补涨
```

### 5. 市值与流通性原则
```
- 流通市值50-200亿：最佳龙头体量
- 太小（<30亿）：容易被控盘，持续性差
- 太大（>500亿）：拉升难度大，难成短线龙头
```

---

## 真实龙头vs补涨vs跟风的区别

| 维度 | 真龙头 | 补涨 | 跟风 |
|------|--------|------|------|
| **涨停时间** | 板块内最早 | 板块确认后 | 板块高潮时 |
| **封单强度** | 强（封单比>5%） | 中等 | 弱（反复开板）|
| **连板预期** | 高（2-3板） | 中（1-2板） | 低（首板） |
| **带动效应** | 带动板块 | 跟随板块 | 被动上涨 |
| **次日溢价** | 高开高走 | 高开低走/平开 | 低开低走 |
| **历史股性** | 活跃，常涨停 | 一般 | 冷门 |
| **消息纯度** | 核心受益 | 间接受益 | 沾边概念 |

```python
def identify_leaders(zt_stocks: list) -> dict:
    """
    从涨停股票中识别龙头股
    
    输入: [{
        'code': '300001',
        'name': '特变电工',
        'zt_time': '10:00:00',
        'block': 'AI应用',
        'seal_amount': 50000000,  # 封单金额
        'float_market_cap': 1000000000,  # 流通市值
        'history_zt_count': 5,  # 近30日涨停次数
        'max_continuous': 2,  # 历史最高连板
    }]
    
    输出: {
        'leaders': [...],  # 龙头股列表
        'supporters': [...],  # 补涨股列表
        'followers': [...],  # 跟风股列表
    }
    """
    # 1. 按板块分组
    blocks = group_by_block(zt_stocks)
    
    for block, stocks in blocks.items():
        # 2. 计算各项指标
        for stock in stocks:
            stock['score'] = calculate_leader_score(stock, stocks)
        
        # 3. 排序识别
        sorted_stocks = sorted(stocks, key=lambda x: x['score'], reverse=True)
        
        # 4. 分类
        leaders.append(sorted_stocks[0])  # 最高分是龙头
        supporters.extend(sorted_stocks[1:3])  # 2-3名是补涨
        followers.extend(sorted_stocks[3:])  # 其余是跟风
```

**评分算法**：
```python
def calculate_leader_score(stock, all_stocks_in_block):
    score = 0
    
    # 1. 涨停时间排名（权重40%）
    time_rank = get_time_rank(stock, all_stocks_in_block)
    score += (1 - time_rank / len(all_stocks_in_block)) * 40
    
    # 2. 封单强度（权重25%）
    seal_ratio = stock['seal_amount'] / stock['float_market_cap']
    score += min(seal_ratio * 100, 25)  # 封顶25分
    
    # 3. 历史股性（权重20%）
    history_score = min(stock['history_zt_count'] * 2 + stock['max_continuous'] * 3, 20)
    score += history_score
    
    # 4. 市值适中度（权重15%）
    cap = stock['float_market_cap'] / 1e8  # 转为亿
    if 50 <= cap <= 200:
        score += 15
    elif 30 <= cap < 50 or 200 < cap <= 500:
        score += 10
    else:
        score += 5
    
    return score
```
