# 债券上攻排行悬停分时图 - 重新设计方案

## 参考样式

**大盘概览股市/债市概览悬浮图表特点：**
- 深色渐变背景：`linear-gradient(145deg, #1a1a2e 0%, #16213e 100%)`
- 边框颜色根据涨跌变化：红色(up)/绿色(down)
- 显示**涨跌幅**而非价格
- 带发光阴影效果
- 显示当前值和涨跌pill
- 图表面积填充渐变色

## 新设计方案

### 一、视觉风格统一

完全复用大盘概览的悬浮图表样式：

| 元素 | 样式 |
|------|------|
| 背景 | `linear-gradient(145deg, #1a1a2e 0%, #16213e 100%)` |
| 边框 | 2px solid，涨跌变色 |
| 尺寸 | 320px × 200px（比大盘略小，适合排行行高）|
| 阴影 | 多层发光效果 |
| 文字 | 白色/灰色分级 |

### 二、显示内容

**主指标：涨跌幅 (change_pct)**
- Y轴显示涨跌幅百分比
- 面积图填充（涨红跌绿）
- 当前涨跌幅大字体显示

**辅助信息：**
- 债券代码 + 名称
- 当前涨跌幅数值
- 数据来源标签（⚡缓存 / 📊数据库）
- 响应时间

### 三、交互设计

参考大盘概览的悬停交互：

```javascript
// 智能定位（优先下方，空间不足则上方）
const spaceBelow = window.innerHeight - rect.bottom - margin;
const spaceAbove = rect.top - margin;

if (spaceBelow >= popupHeight) {
    top = rect.bottom + margin;  // 下方
} else if (spaceAbove >= popupHeight) {
    top = rect.top - popupHeight - margin;  // 上方
}

// 水平居中于行
left = rect.left + (rect.width - popupWidth) / 2;
```

**防抖策略：**
- 100ms延迟触发（避免快速滑动频繁请求）
- 鼠标移出100ms后关闭（可移入浮层保持打开）

#### 4.3 JavaScript实现

```javascript
// 债券排行悬浮分时图
let _bondTickChart = null;
let _bondTickHover = { row: false, popup: false };
let _bondTickHideTimer = null;
let _bondTickLoadTimer = null;

async function showBondTickPopup(bondCode, bondName, event) {
    if (bondCode === 'none' || !bondCode) return;
    
    _bondTickHover.row = true;
    clearTimeout(_bondTickHideTimer);
    
    const popup = document.getElementById('bond-tick-popup');
    if (!popup || popup.style.display === 'block') return;
    
    // 智能定位
    const rect = event.target.closest('tr').getBoundingClientRect();
    const popupWidth = 320;
    const popupHeight = 200;
    const margin = 10;
    
    const spaceBelow = window.innerHeight - rect.bottom - margin;
    const spaceAbove = rect.top - margin;
    
    let top;
    if (spaceBelow >= popupHeight) {
        top = rect.bottom + margin;
    } else if (spaceAbove >= popupHeight) {
        top = rect.top - popupHeight - margin;
    } else {
        top = (window.innerHeight - popupHeight) / 2;
    }
    
    let left = rect.left + (rect.width - popupWidth) / 2;
    if (left < margin) left = margin;
    if (left + popupWidth > window.innerWidth - margin) {
        left = window.innerWidth - popupWidth - margin;
    }
    
    popup.style.left = left + 'px';
    popup.style.top = top + 'px';
    popup.style.display = 'block';
    
    // 显示加载中
    document.getElementById('bond-tick-name').textContent = bondName || bondCode;
    document.getElementById('bond-tick-code').textContent = bondCode;
    document.getElementById('bond-tick-change').textContent = '--';
    document.getElementById('bond-tick-source').textContent = '加载中...';
    document.getElementById('bond-tick-source').className = 'bond-tick-source';
    
    // 防抖加载
    clearTimeout(_bondTickLoadTimer);
    _bondTickLoadTimer = setTimeout(() => loadBondTickData(bondCode), 100);
}

async function loadBondTickData(bondCode) {
    const startTime = Date.now();
    
    try {
        const response = await fetch(`/api/bond/tick/${bondCode}`);
        const result = await response.json();
        const loadTime = Date.now() - startTime;
        
        if (!result.success || result.data.length === 0) {
            showBondTickError('无数据');
            return;
        }
        
        // 显示来源
        const isRedis = result.source === 'redis';
        const sourceEl = document.getElementById('bond-tick-source');
        sourceEl.textContent = isRedis ? `⚡缓存 ${loadTime}ms` : `📊数据库 ${loadTime}ms`;
        sourceEl.className = 'bond-tick-source ' + (isRedis ? 'redis' : 'mysql');
        
        // 渲染图表
        renderBondTickChart(result.data);
        
    } catch (e) {
        showBondTickError('加载失败');
    }
}

function renderBondTickChart(data) {
    const chartDom = document.getElementById('bond-tick-chart');
    if (!_bondTickChart) {
        _bondTickChart = echarts.init(chartDom);
    }
    
    // 取涨跌幅数据
    const times = data.map(d => d.time?.slice(0,5) || '');
    const changes = data.map(d => d.change_pct || 0);
    
    const firstVal = changes[0] || 0;
    const lastVal = changes[changes.length - 1] || 0;
    const isUp = lastVal >= firstVal;
    
    // 更新边框颜色
    const popup = document.getElementById('bond-tick-popup');
    popup.classList.remove('up', 'down');
    popup.classList.add(isUp ? 'up' : 'down');
    
    // 更新当前涨跌幅显示
    const changeEl = document.getElementById('bond-tick-change');
    changeEl.textContent = (lastVal >= 0 ? '+' : '') + lastVal.toFixed(2) + '%';
    changeEl.className = 'bond-tick-change ' + (lastVal >= 0 ? 'up' : 'down');
    
    // 图表颜色
    const lineColor = isUp ? '#ff4757' : '#00d084';
    const areaColor = isUp ?
        { type: 'linear', x: 0, y: 0, x2: 0, y2: 1,
          colorStops: [{ offset: 0, color: 'rgba(255,71,87,0.3)' },
                       { offset: 1, color: 'rgba(255,71,87,0)' }] } :
        { type: 'linear', x: 0, y: 0, x2: 0, y2: 1,
          colorStops: [{ offset: 0, color: 'rgba(0,208,132,0.3)' },
                       { offset: 1, color: 'rgba(0,208,132,0)' }] };
    
    const option = {
        backgroundColor: 'transparent',
        grid: { top: 5, right: 5, bottom: 20, left: 40 },
        tooltip: {
            trigger: 'axis',
            backgroundColor: 'rgba(30, 35, 60, 0.9)',
            borderColor: isUp ? '#ff4757' : '#00d084',
            borderWidth: 1,
            textStyle: { color: '#fff', fontSize: 11 },
            formatter: function(params) {
                const data = params[0];
                const time = data.name;
                const value = data.value;
                const color = value >= 0 ? '#ff4757' : '#00d084';
                return `<div style="font-weight:bold;margin-bottom:4px;">${time}</div>
                        <div style="color:${color}">涨跌幅: ${value >= 0 ? '+' : ''}${value.toFixed(2)}%</div>`;
            }
        },
        xAxis: {
            type: 'category',
            data: times,
            axisLine: { lineStyle: { color: '#8892b0' } },
            axisLabel: { color: '#8892b0', fontSize: 9 }
        },
        yAxis: {
            type: 'value',
            axisLine: { lineStyle: { color: '#8892b0' } },
            splitLine: { lineStyle: { color: 'rgba(255,255,255,0.05)' } },
            axisLabel: { 
                color: '#8892b0', 
                fontSize: 9, 
                formatter: v => v.toFixed(2) + '%'
            }
        },
        series: [{
            type: 'line',
            data: changes,
            smooth: false,
            symbol: 'none',
            lineStyle: { width: 2, color: lineColor },
            areaStyle: { color: areaColor }
        }]
    };
    
    _bondTickChart.setOption(option);
}

function showBondTickError(msg) {
    document.getElementById('bond-tick-source').textContent = msg;
    document.getElementById('bond-tick-source').className = 'bond-tick-source';
    document.getElementById('bond-tick-change').textContent = '--';
    if (_bondTickChart) {
        _bondTickChart.clear();
    }
}

function hideBondTickPopup() {
    _bondTickHover.row = false;
    tryHideBondTick();
}

function onBondTickEnter() {
    _bondTickHover.popup = true;
    clearTimeout(_bondTickHideTimer);
}

function onBondTickLeave() {
    _bondTickHover.popup = false;
    tryHideBondTick();
}

function tryHideBondTick() {
    _bondTickHideTimer = setTimeout(() => {
        if (!_bondTickHover.row && !_bondTickHover.popup) {
            const popup = document.getElementById('bond-tick-popup');
            if (popup) {
                popup.style.display = 'none';
                popup.classList.remove('up', 'down');
            }
            if (_bondTickChart) {
                _bondTickChart.clear();
            }
        }
    }, 100);
}
```
