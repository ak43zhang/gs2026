## 一、核心设计原则

**目标**：新增一个JSON字段，只需要修改**1个文件**（`monitor_bond.py`），其他文件**自动**支持。

### 设计对比

| 方案 | 修改文件数 | 维护成本 | 说明 |
|------|-----------|---------|------|
| 当前方案 | 3个 | 高 | monitor_bond + compute_engine + backfill_json_fields |
| **v3方案** | **1个** | **低** | **只在monitor_bond.py新增计算函数** |

### 实现机制

**关键洞察**：`compute_engine.py` 已经通过 `_import_unified_functions()` 强制从 `monitor_bond.py` 导入计算函数。

```python
# compute_engine.py 已有机制
def _import_unified_functions():
    """强制从 monitor_bond.py 导入统一计算函数"""
    from monitor_bond import (
        compute_ext_indicators,  # 已有
        # 新增字段的计算函数会自动被导入
    )
    return locals()
```

**扩展思路**：
1. 在 `monitor_bond.py` 中定义计算函数（已有）
2. 在 `monitor_bond.py` 中定义**字段注册表**（新增）
3. `compute_engine.py` 读取注册表，**自动生成**回填方法
4. `backfill_json_fields.py` 读取注册表，**自动识别**可回填字段

```python
def compute_mkt_shape(dependencies: dict, history: list) -> str:
    """
    计算大盘形态（实时+回填共用）
    
    5种形态：单边上行/单边下行/低开高走/高开低走/横盘
    
    Args:
        dependencies: {'mkt_vs_open_pct': float}
        history: [{'mkt_vs_open_pct': float, 'time_sec': int}, ...]
    
    Returns:
        形态名: '单边上行'/'单边下行'/'低开高走'/'高开低走'/'横盘'
    """
    if len(history) < 10:
        return '横盘'
    
    vs_values = [h.get('mkt_vs_open_pct', 0) for h in history]
    cur_pct = dependencies.get('mkt_vs_open_pct', 0)
    vs_values.append(cur_pct)
    
    high_pct = max(vs_values)
    low_pct = min(vs_values)
    total_range = high_pct - low_pct
    
    if total_range < 0.5:
        return '横盘'
    
    n = len(vs_values)
    high_idx = vs_values.index(high_pct)
    low_idx = vs_values.index(low_pct)
    high_pos = high_idx / (n - 1) if n > 1 else 0
    low_pos = low_idx / (n - 1) if n > 1 else 0
    
    sig = total_range * 0.5 if total_range > 0 else 0.001
    recovery = cur_pct - low_pct
    drawdown = high_pct - cur_pct
    
    if low_pos < 0.2 and cur_pct > 0 and drawdown < sig:
        return '单边上行'
    if high_pos < 0.2 and cur_pct < 0 and recovery < sig:
        return '单边下行'
    if low_pos > 0.2 and recovery > sig and cur_pct > 0:
        return '低开高走'
    if high_pos > 0.2 and drawdown > sig and cur_pct < 0:
        return '高开低走'
    
    return '横盘'


def compute_mkt_shape_detail(dependencies: dict, history: list) -> str:
    """
    计算大盘形态详情（实时+回填共用）
    """
    if len(history) < 10:
        return '数据不足'
    
    vs_values = [h.get('mkt_vs_open_pct', 0) for h in history]
    cur_pct = dependencies.get('mkt_vs_open_pct', 0)
    vs_values.append(cur_pct)
    
    high_pct = max(vs_values)
    low_pct = min(vs_values)
    total_range = high_pct - low_pct
    
    n = len(vs_values)
    high_idx = vs_values.index(high_pct)
    low_idx = vs_values.index(low_pct)
    high_pos = high_idx / (n - 1) if n > 1 else 0
    low_pos = low_idx / (n - 1) if n > 1 else 0
    
    sig = total_range * 0.5 if total_range > 0 else 0.001
    recovery = cur_pct - low_pct
    drawdown = high_pct - cur_pct
    
    if total_range < 0.5:
        return f'振幅{total_range:.2f}%<0.5%'
    if low_pos < 0.2 and cur_pct > 0 and drawdown < sig:
        return f'开盘即最低，当前{cur_pct:+.2f}%'
    if high_pos < 0.2 and cur_pct < 0 and recovery < sig:
        return f'开盘即最高，当前{cur_pct:+.2f}%'
    if low_pos > 0.2 and recovery > sig and cur_pct > 0:
        return f'低点在{low_pos:.0%}，回升{recovery:.2f}%'
    if high_pos > 0.2 and drawdown > sig and cur_pct < 0:
        return f'高点在{high_pos:.0%}，回落{drawdown:.2f}%'
    
    return f'高{high_pct:+.2f}% 低{low_pct:+.2f}% 收{cur_pct:+.2f}%'
```

```python
def get_json_field_registry():
    """
    获取JSON字段注册表
    
    供 compute_engine.py 和 backfill_json_fields.py 自动读取
    """
    return JSON_FIELD_REGISTRY


def compute_json_field(field_name: str, dependencies: dict, history: list = None) -> any:
    """
    通用JSON字段计算接口
    
    供 compute_engine.py 统一调用
    
    Args:
        field_name: 字段名
        dependencies: 依赖字段值 {'mkt_vs_open_pct': 1.5, ...}
        history: 历史数据（如果needs_history=True）
    
    Returns:
        字段值
    """
    registry = get_json_field_registry()
    if field_name not in registry:
        raise ValueError(f"未知字段: {field_name}")
    
    field_config = registry[field_name]
    computer_name = field_config['computer']
    
    # 获取计算函数
    computer = globals().get(computer_name)
    if computer is None:
        raise ValueError(f"计算函数不存在: {computer_name}")
    
    # 调用计算
    if field_config.get('needs_history') and history is not None:
        return computer(dependencies, history)
    else:
        return computer(dependencies)
```

```python
# compute_engine.py 已有代码（无需修改）
class ComputeEngine:
    def __init__(self):
        # ... 原有初始化 ...
        
        # 自动从 monitor_bond.py 导入字段注册表
        from monitor_bond import get_json_field_registry
        self.json_field_registry = get_json_field_registry()
        
        # 自动初始化注册表中声明的状态变量
        for field_name, config in self.json_field_registry.items():
            for state_var in config.get('state_vars', []):
                if state_var == '_mkt_shape_history':
                    self.mkt_shape_history = []
    
    def _compute_json_field(self, field_name: str, dependencies: dict) -> any:
        """
        通用JSON字段计算（自动读取注册表）
        """
        from monitor_bond import compute_json_field
        
        # 构建历史数据
        history = []
        if field_name in self.json_field_registry:
            config = self.json_field_registry[field_name]
            if config.get('needs_history'):
                # 从历史缓存构建
                history = self.mkt_shape_history
        
        return compute_json_field(field_name, dependencies, history)
    
    def _update_json_field_state(self, field_name: str, deps: dict):
        """
        更新JSON字段状态（供skip-existing模式使用）
        """
        if field_name not in self.json_field_registry:
            return
        
        config = self.json_field_registry[field_name]
        
        # 更新依赖的历史状态
        if 'mkt_vs_open_pct' in deps:
            for state_var in config.get('state_vars', []):
                if state_var == '_mkt_shape_history':
                    self.mkt_shape_history.append({
                        'mkt_vs_open_pct': deps['mkt_vs_open_pct']
                    })
```

## 七、总结

### 核心设计

**"一次修改到处调用"** 的实现：

1. **计算函数定义在 `monitor_bond.py`**（一处定义）
2. **字段注册表在 `monitor_bond.py`**（一处注册）
3. **`compute_engine.py` 自动读取注册表**（零修改）
4. **`backfill_json_fields.py` 自动读取注册表**（零修改）

### 优势

| 优势 | 说明 |
|------|------|
| **维护简单** | 新增字段只需修改1个文件 |
| **一致性保证** | 实时和回填共用同一套计算逻辑 |
| **扩展容易** | 添加新字段只需3步：状态+函数+注册 |
| **错误减少** | 不需要在多个文件中同步修改 |

### 与之前方案的对比

| 方案 | 修改文件数 | 新增字段工作量 |
|------|-----------|-------------|
| 旧方案 | 3个 | 高（多处同步） |
| v3方案 | **1个** | **低（一处定义）** |

## 三、实施方案（基于JSON字段快速回填方案v3）

### 3.1 核心设计原则

**遵循JSON字段快速回填方案v3的设计原则**：
1. **计算逻辑统一**：所有计算函数定义在 `monitor_bond.py`，实时和回填共用
2. **字段级独立更新**：使用 `JSON_SET()` 只修改指定key，不影响其他字段
3. **默认覆盖**：无条件覆盖指定key的value
4. **复用现有架构**：回填通过 `compute_engine.py` 调用相同的函数

### 3.2 修改文件清单

| 文件 | 修改内容 | 必需 |

```python
def compute_mkt_shape(dependencies: dict, history: list) -> str:
    """
    计算大盘形态（实时+回填共用）
    
    5种形态：
        - 单边上行: 开盘即最低，当前>0，回落不显著
        - 单边下行: 开盘即最高，当前<0，回升不显著
        - 低开高走: 低点不在开头，回升显著，当前>0
        - 高开低走: 高点不在开头，回落显著，当前<0
        - 横盘: 振幅<0.5%或其他条件不满足
    
    Args:
        dependencies: {'mkt_vs_open_pct': float}
        history: [{'mkt_vs_open_pct': float, 'time_sec': int}, ...]
    
    Returns:
        形态名: '单边上行'/'单边下行'/'低开高走'/'高开低走'/'横盘'
    """
    if len(history) < 10:
        return '横盘'
    
    # 合并历史和当前
    vs_values = [h.get('mkt_vs_open_pct', 0) for h in history]
    cur_pct = dependencies.get('mkt_vs_open_pct', 0)
    vs_values.append(cur_pct)
    
    high_pct = max(vs_values)
    low_pct = min(vs_values)
    total_range = high_pct - low_pct
    
    # 横盘判定
    if total_range < 0.5:
        return '横盘'
    
    # 位置计算
    n = len(vs_values)
    high_idx = vs_values.index(high_pct)
    low_idx = vs_values.index(low_pct)
    high_pos = high_idx / (n - 1) if n > 1 else 0
    low_pos = low_idx / (n - 1) if n > 1 else 0
    
    # 阈值
    sig = total_range * 0.5 if total_range > 0 else 0.001
    recovery = cur_pct - low_pct
    drawdown = high_pct - cur_pct
    
    # 形态判定
    if low_pos < 0.2 and cur_pct > 0 and drawdown < sig:
        return '单边上行'
    if high_pos < 0.2 and cur_pct < 0 and recovery < sig:
        return '单边下行'
    if low_pos > 0.2 and recovery > sig and cur_pct > 0:
        return '低开高走'
    if high_pos > 0.2 and drawdown > sig and cur_pct < 0:
        return '高开低走'
    
    return '横盘'


def compute_mkt_shape_detail(dependencies: dict, history: list) -> str:
    """
    计算大盘形态详情（实时+回填共用）
    
    Args:
        dependencies: {'mkt_vs_open_pct': float}
        history: [{'mkt_vs_open_pct': float, 'time_sec': int}, ...]
    
    Returns:
        形态详细说明
    """
    if len(history) < 10:
        return '数据不足'
    
    vs_values = [h.get('mkt_vs_open_pct', 0) for h in history]
    cur_pct = dependencies.get('mkt_vs_open_pct', 0)
    vs_values.append(cur_pct)
    
    high_pct = max(vs_values)
    low_pct = min(vs_values)
    total_range = high_pct - low_pct
    
    n = len(vs_values)
    high_idx = vs_values.index(high_pct)
    low_idx = vs_values.index(low_pct)
    high_pos = high_idx / (n - 1) if n > 1 else 0
    low_pos = low_idx / (n - 1) if n > 1 else 0
    
    sig = total_range * 0.5 if total_range > 0 else 0.001
    recovery = cur_pct - low_pct
    drawdown = high_pct - cur_pct
    
    if total_range < 0.5:
        return f'振幅{total_range:.2f}%<0.5%'
    if low_pos < 0.2 and cur_pct > 0 and drawdown < sig:
        return f'开盘即最低，当前{cur_pct:+.2f}%'
    if high_pos < 0.2 and cur_pct < 0 and recovery < sig:
        return f'开盘即最高，当前{cur_pct:+.2f}%'
    if low_pos > 0.2 and recovery > sig and cur_pct > 0:
        return f'低点在{low_pos:.0%}，回升{recovery:.2f}%'
    if high_pos > 0.2 and drawdown > sig and cur_pct < 0:
        return f'高点在{high_pos:.0%}，回落{drawdown:.2f}%'
    
    return f'高{high_pct:+.2f}% 低{low_pct:+.2f}% 收{cur_pct:+.2f}%'
```

## 九、总结

### 核心设计

1. **计算逻辑统一**：`compute_mkt_shape` 定义在 `monitor_bond.py`，实时和回填共用
