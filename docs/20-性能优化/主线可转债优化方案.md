## 1. 主线可转债优化方案-最终版


## 确定需求

1. ✅ 比较已分析债券数量
2. ✅ 无需持久化，内存缓存即可
3. ✅ 无需强制刷新机制
4. ✅ 全量刷新，非增量

## 最终方案

### 核心逻辑

```python
class MainlineBondsService:
    def __init__(self):
        self._last_analyzed_count = 0  # 上次已分析数量
    
    def refresh_mainline_bonds(self):
        """
        刷新主线可转债
        规则：只有已分析数量变化时才刷新
        """
        # 1. 获取当前数据
        current_bonds = self._fetch_bonds_data()
        
        # 2. 统计已分析数量
        analyzed_count = sum(1 for b in current_bonds if b.get('is_analyzed', False))
        
        # 3. 对比数量
        if analyzed_count == self._last_analyzed_count:
            logger.info(f"[主线可转债] 已分析数量未变 ({analyzed_count}只)，跳过刷新")
            return False
        
        # 4. 数量变化，执行刷新
        logger.info(f"[主线可转债] 已分析数量变化 {self._last_analyzed_count}→{analyzed_count}，执行刷新")
        self._last_analyzed_count = analyzed_count
        
        # 5. 刷新界面
        self._update_ui(current_bonds)
        return True
```

### 定时任务调用

def scheduled_refresh():

## 2. 主线可转债优化方案


## 核心问题

当前逻辑：每次定时刷新都重新获取数据并刷新界面
优化目标：数据未变化时不刷新，减少刷新次数，提高效率

## 设计思路

```
┌─────────────┐     ┌─────────────┐     ┌─────────────┐
│   获取数据   │────→│  对比数量   │────→│  决定是否刷新 │
└─────────────┘     └─────────────┘     └─────────────┘
                           │
                    数量变化？
                      /    \
                   是        否
                   /          \
              ┌──────┐      ┌──────┐
              │ 刷新 │      │ 跳过 │
              │ 界面 │      │ 保持 │
              └──────┘      └──────┘
```

## 具体实现

### 1. 数据缓存与对比

```python
# 在类中添加缓存
class MainlineBondsService:
    def __init__(self):
        self._last_bond_count = 0
        self._last_bonds_data = None
        self._last_update_time = None
    
    def refresh_if_changed(self):
        """只有数据变化时才刷新"""
        # 获取当前数据
        current_bonds = self.fetch_bonds_data()
        current_count = len(current_bonds)
        
        # 对比数量
        if current_count == self._last_bond_count:
            # 数量相同，进一步对比内容（可选）
            if self._is_data_identical(current_bonds, self._last_bonds_data):
                logger.info(f"[主线可转债] 数据未变化 ({current_count}只)，跳过刷新")
                return False  # 不刷新
        
        # 数据变化，更新缓存并刷新
        self._last_bond_count = current_count
        self._last_bonds_data = current_bonds
        self._last_update_time = datetime.now()
        
        logger.info(f"[主线可转债] 数据变化 {self._last_bond_count}→{current_count}，执行刷新")
        self.refresh_ui(current_bonds)
        return True  # 已刷新
    
    def _is_data_identical(self, new_data, old_data):
        """深度对比数据是否相同（可选，用于精确判断）"""
        if old_data is None:
            return False
        
        # 简单对比：对比前N只的code
        new_codes = {b['bond_code'] for b in new_data[:20]}
        old_codes = {b['bond_code'] for b in old_data[:20]}
        return new_codes == old_codes
```

### 2. 前端优化

```javascript
// 前端也做缓存，避免重复渲染
class MainlineBondsStore {
    constructor() {
        this.lastData = null;
        this.lastCount = 0;
    }
    
    updateIfChanged(newData) {
        const newCount = newData.length;
        
        // 数量相同，检查内容
        if (newCount === this.lastCount) {
            // 对比关键字段
            const changed = this._hasSignificantChange(newData, this.lastData);
            if (!changed) {
                console.log('[主线可转债] 数据未变化，跳过渲染');
                return false;
            }
        }
        
        // 更新并渲染
        this.lastData = newData;
        this.lastCount = newCount;
        this.render(newData);
        return true;
    }
    
    _hasSignificantChange(newData, oldData) {
        if (!oldData) return true;
        
        // 对比前5只债券的code和price
        for (let i = 0; i < Math.min(5, newData.length); i++) {
            if (newData[i].bond_code !== oldData[i]?.bond_code) return true;
            if (newData[i].price !== oldData[i]?.price) return true;
        }
        return false;
    }
}
```

### 3. 定时刷新优化

```python
# 优化前：固定间隔刷新
def scheduled_refresh():
    while True:
        time.sleep(60)  # 每分钟
        refresh_mainline_bonds()  # 总是刷新

# 优化后：智能刷新
def smart_refresh():
    service = MainlineBondsService()
    
    while True:
        time.sleep(60)  # 每分钟检查
        
        # 只有数据变化才刷新
        if service.refresh_if_changed():
            logger.info("界面已更新")
        else:
            logger.info("数据未变，跳过")
```
