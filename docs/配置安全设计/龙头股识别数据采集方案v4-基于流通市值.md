### 2.3 龙头细分逻辑（基于流通市值）

```python
def classify_leader_by_market_cap(stock_code: str, trade_date: str) -> str:
    """
    根据流通市值判断龙头细分类型
    
    Args:
        stock_code: 股票代码
        trade_date: 交易日期YYYYMMDD
        
    Returns:
        '权重龙头' / '板块中军' / '小盘情绪龙' / '未知'
    """
    # 查询该股票当天的流通市值
    sql = """
        SELECT circulating_market_cap 
        FROM data_stock_spot_em 
        WHERE stock_code = :stock_code AND trade_date = :trade_date
        LIMIT 1
    """
    
    with engine.connect() as conn:
        result = conn.execute(text(sql), {
            'stock_code': stock_code,
            'trade_date': trade_date
        })
        row = result.fetchone()
        
        if not row or row[0] is None:
            return '未知'
        
        market_cap = float(row[0])
        
        # 根据市值判断类型
        if market_cap > 500:
            return '权重龙头'
        elif 200 <= market_cap <= 500:
            return '板块中军'
        else:
            return '小盘情绪龙'
```

### 2.4 整合到龙头识别流程

```python
def get_leader_display_role(stock_code: str, mainline_name: str, 
                            trading_date: str, ai_role: str) -> dict:
    """
    获取龙头展示role（含细分类型）
    
    逻辑：
    1. 如果AI已定义细分role（权重龙头/板块中军/小盘情绪龙），直接返回
    2. 如果AI只定义了'龙头'，根据流通市值细分
    3. 如果AI未定义龙头，返回原role
    """
    result = {
        'original_role': ai_role,
        'display_role': ai_role,
        'market_cap': None,
        'leader_type': None  # 权重龙头/板块中军/小盘情绪龙
    }
    
    # 如果AI已定义细分类型，直接返回
    if ai_role in ['权重龙头', '板块中军', '小盘情绪龙']:
        result['leader_type'] = ai_role
        return result
    
    # 如果AI只定义了'龙头'，根据市值细分
    if ai_role == '龙头':
        leader_type = classify_leader_by_market_cap(stock_code, trading_date)
        result['display_role'] = leader_type
        result['leader_type'] = leader_type
        
        # 获取市值用于展示
        sql = """
            SELECT circulating_market_cap 
            FROM data_stock_spot_em 
            WHERE stock_code = :stock_code AND trade_date = :trade_date
        """
        with engine.connect() as conn:
            row = conn.execute(text(sql), {
                'stock_code': stock_code,
                'trade_date': trading_date
            }).fetchone()
            if row:
                result['market_cap'] = row[0]
    
    return result
```
