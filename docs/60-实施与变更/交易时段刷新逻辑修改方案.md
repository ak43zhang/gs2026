## 一、当前实现分析

### 当前交易时段判断

```javascript
function isTradingHours() {
    const timeVal = getCurrentTimeVal();
    // 早盘: 09:30-11:30, 午盘: 13:00-15:00
    const isMorning = (timeVal >= 930 && timeVal <= 1130);
    const isAfternoon = (timeVal >= 1300 && timeVal <= 1500);
    return isMorning || isAfternoon;  // 15:00整点是交易时段
}

function isAuctionHours() {
    // 集合竞价: 09:25-09:30 (只加载大盘概览,成功即停)
    const timeVal = getCurrentTimeVal();
    return (timeVal >= 925 && timeVal < 930);
}

function isNonTradingHours() {
    // 非交易时段: 00:00-09:25, 11:30-13:00, 15:00-24:00
    const timeVal = getCurrentTimeVal();
    return (timeVal < 925) ||                    // 早盘集合竞价前
           (timeVal > 1130 && timeVal < 1300) || // 午休
           (timeVal > 1500);                     // 15:00后
}
```

### 当前行为

| 时间段 | 状态 | 刷新行为 |
|--------|------|----------|
| 00:00-09:25 | 非交易 | ❌ 不刷新 |
| 09:25-09:30 | 集合竞价 | ✅ 只加载大盘概览一次 |
| 09:30-11:30 | 交易 | ✅ 正常刷新(3秒) |
| 11:30-13:00 | 午休 | ❌ 不刷新 |
| 13:00-15:00 | 交易 | ✅ 正常刷新(3秒) |
| **15:00-24:00** | **非交易** | **❌ 不刷新** |

### 预设修改方案（基于常见需求推测）

#### 方案甲：15:00后继续刷新到15:30

```javascript
function isTradingHours() {
    const timeVal = getCurrentTimeVal();
    const isMorning = (timeVal >= 930 && timeVal <= 1130);
    const isAfternoon = (timeVal >= 1300 && timeVal <= 1530);  // 延长到15:30
    return isMorning || isAfternoon;
}

function isNonTradingHours() {
    const timeVal = getCurrentTimeVal();
    return (timeVal < 925) ||
           (timeVal > 1130 && timeVal < 1300) ||
           (timeVal > 1530);  // 15:30后才算非交易
}
```

#### 方案乙：15:00后永不停止（只要有数据）

```javascript
function isTradingHours() {
    // 只要有数据就认为是交易时段（15:00后数据仍在更新）
    // 通过检查是否有最新数据来判断
    return true;  // 简化：永远刷新，依赖后端数据是否存在
}

// 或者更精细：15:00后降低频率（30秒一次）
function getRefreshInterval() {
    const timeVal = getCurrentTimeVal();
    if (timeVal > 1500) return 30000;  // 15:00后30秒一次
    return 3000;  // 正常3秒
}
```
