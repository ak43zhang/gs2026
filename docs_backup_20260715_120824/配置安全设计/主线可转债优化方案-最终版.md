# 主线可转债优化方案（最终版）

## 确定需求

1. ✅ 比较已分析债券数量
2. ✅ 无需持久化，内存缓存即可
3. ✅ 无需强制刷新机制
4. ✅ 全量刷新，非增量

## 最终方案

### 核心逻辑

```python
class MainlineBondsService:
    def __init__(self):
        self._last_analyzed_count = 0  # 上次已分析数量
    
    def refresh_mainline_bonds(self):
        """
        刷新主线可转债
        规则：只有已分析数量变化时才刷新
        """
        # 1. 获取当前数据
        current_bonds = self._fetch_bonds_data()
        
        # 2. 统计已分析数量
        analyzed_count = sum(1 for b in current_bonds if b.get('is_analyzed', False))
        
        # 3. 对比数量
        if analyzed_count == self._last_analyzed_count:
            logger.info(f"[主线可转债] 已分析数量未变 ({analyzed_count}只)，跳过刷新")
            return False
        
        # 4. 数量变化，执行刷新
        logger.info(f"[主线可转债] 已分析数量变化 {self._last_analyzed_count}→{analyzed_count}，执行刷新")
        self._last_analyzed_count = analyzed_count
        
        # 5. 刷新界面
        self._update_ui(current_bonds)
        return True
```

```python
class MainlineBondsService:
    """主线可转债服务 - 智能刷新版"""
    
    def __init__(self):
        self._last_analyzed_count = 0
        self._logger = logging.getLogger(__name__)
    
    def refresh(self) -> bool:
        """
        智能刷新：只有已分析数量变化时才刷新
        
        Returns:
            bool: True=已刷新, False=跳过
        """
        # 获取当前数据
        bonds = self._fetch_from_db()
        
        # 统计已分析数量
        analyzed_count = sum(1 for b in bonds if b.get('analyzed_at'))
        
        # 对比
        if analyzed_count == self._last_analyzed_count:
            self._logger.info(f"[主线可转债] 已分析数量未变 ({analyzed_count}只)，跳过刷新")
            return False
        
        # 执行刷新
        self._logger.info(f"[主线可转债] 已分析数量变化 {self._last_analyzed_count}→{analyzed_count}")
        self._last_analyzed_count = analyzed_count
        
        # 更新界面
        self._push_to_frontend(bonds)
        return True
```
