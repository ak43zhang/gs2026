class LeaderAnalyzer:
    """龙头分析器（新版）"""
    
    def analyze(self, anomaly: dict) -> dict:
        """
        分析单只股票的龙头地位
        
        Args:
            anomaly: stock_anomaly表的一条记录
        
        Returns:
            包含所有维度的分析结果
        """
        # 1. 获取基础数据
        stock_code = anomaly['stock_code']
        stock_name = anomaly['stock_name']
        
        # 2. 关联stock_basic获取市值
        market_cap = self.get_market_cap(stock_code)
        
        # 3. 构建Prompt
        prompt = self.build_prompt(anomaly, market_cap)
        
        # 4. AI分析
        response = self.call_ai(prompt)
        
        # 5. 解析结果
        analysis = json.loads(response)
        
        # 6. 组装完整记录（包含所有维度）
        record = {
            # 基础字段
            'anomaly_id': anomaly['id'],
            'trading_date': anomaly['trading_date'],
            'stock_code': stock_code,
            'stock_name': stock_name,
            'anomaly_time': anomaly['anomaly_time'],
            'price': anomaly['price'],
            'change_pct': anomaly['change_pct'],
            'continuous_zt': anomaly['continuous_zt'],
            'mainline_names': anomaly['mainline_names'],
            
            # 龙头维度
            'leader_status': analysis['龙头识别']['leader_status'],
            'leader_score': analysis['龙头识别']['leader_score'],
            'mainline_rank': analysis['龙头识别']['mainline_rank'],
            'status_reason': analysis['龙头识别']['status_reason'],
            'continuous_expect': analysis['龙头识别']['continuous_expect'],
            'next_day_expect': analysis['龙头识别']['next_day_expect'],
            
            # 体量维度
            'size_type': analysis['体量分类']['size_type'],
            'float_market_cap': analysis['体量分类']['float_market_cap'],
            'strategy': analysis['体量分类']['strategy'],
            'risk_level': analysis['体量分类']['risk_level'],
            'fit_for': analysis['体量分类']['fit_for'],
            
            # 扩展字段（当前为空）
            'ext_field_1': json.dumps(analysis['扩展字段']['ext_field_1']),
            'ext_field_2': json.dumps(analysis['扩展字段']['ext_field_2']),
        }
        
        return record
    
    def save(self, record: dict):
        """保存到宽表（单表写入）"""
        engine = create_engine(config_util.get_config('common.url'))
        
        with engine.connect() as conn:
            # 使用INSERT ... ON DUPLICATE KEY UPDATE
            sql = text("""
                INSERT INTO stock_anomaly_analysis (
                    anomaly_id, trading_date, stock_code, stock_name,
                    anomaly_time, price, change_pct, continuous_zt, mainline_names,
                    leader_status, leader_score, mainline_rank, status_reason,
                    continuous_expect, next_day_expect,
                    size_type, float_market_cap, strategy, risk_level, fit_for,
                    ext_field_1, ext_field_2
                ) VALUES (
                    :anomaly_id, :trading_date, :stock_code, :stock_name,
                    :anomaly_time, :price, :change_pct, :continuous_zt, :mainline_names,
                    :leader_status, :leader_score, :mainline_rank, :status_reason,
                    :continuous_expect, :next_day_expect,
                    :size_type, :float_market_cap, :strategy, :risk_level, :fit_for,
                    :ext_field_1, :ext_field_2
                )
                ON DUPLICATE KEY UPDATE
                    leader_status = VALUES(leader_status),
                    leader_score = VALUES(leader_score),
                    status_reason = VALUES(status_reason),
                    size_type = VALUES(size_type),
                    strategy = VALUES(strategy),
                    updated_at = NOW()
            """)
            
            conn.execute(sql, record)
            conn.commit()

def analyze_all_leaders(date: str = None):
    """
    批量分析某日所有涨停股票的龙头地位
    
    Args:
        date: 交易日期，默认今天
    """
    if date is None:
        date = datetime.now().strftime('%Y-%m-%d')
    
    logger.info(f"[龙头分析] 开始分析 {date}")
    
    # 1. 获取当日所有涨停股票
    engine = create_engine(config_util.get_config('common.url'))
    
    with engine.connect() as conn:
        df = pd.read_sql(f"""
            SELECT * FROM stock_anomaly
            WHERE trading_date = '{date}'
              AND anomaly_type = 'zt_hit'
            ORDER BY anomaly_time
        """, conn)
    
    logger.info(f"[龙头分析] 共 {len(df)} 只涨停股票")
    
    # 2. 逐个分析
    analyzer = LeaderAnalyzer()
    
    for _, row in df.iterrows():
        try:
            # 分析
            record = analyzer.analyze(row.to_dict())
            
            # 保存
            analyzer.save(record)
            
            logger.info(f"[龙头分析] {record['stock_code']} {record['stock_name']} | "
                       f"{record['leader_status']} | {record['size_type']} | "
                       f"评分{record['leader_score']}")
            
        except Exception as e:
            logger.error(f"[龙头分析] {row['stock_code']} 分析失败: {e}")
            continue
    
    logger.info(f"[龙头分析] 完成")
```

class LeaderService:
    """龙头查询服务"""
    
    def get_leaders(self, date: str, mainline: str = None, 
                   size_type: str = None, min_score: int = 0):
        """
        查询龙头列表（单表查询）
        
        Args:
            date: 交易日期
            mainline: 主线名称（可选）
            size_type: 体量类型（可选）
            min_score: 最低评分（可选）
        """
        engine = create_engine(config_util.get_config('common.url'))
        
        # 构建查询（单表，无需JOIN）
        sql = """
            SELECT * FROM stock_anomaly_analysis
            WHERE trading_date = :date
              AND leader_status = '龙头'
        """
        
        params = {'date': date}
        
        if mainline:
            sql += " AND mainline_names LIKE :mainline"
            params['mainline'] = f'%{mainline}%'
        
        if size_type:
            sql += " AND size_type = :size_type"
            params['size_type'] = size_type
        
        if min_score > 0:
            sql += " AND leader_score >= :min_score"
            params['min_score'] = min_score
        
        sql += " ORDER BY leader_score DESC"
        
        with engine.connect() as conn:
            df = pd.read_sql(text(sql), conn, params=params)
        
        return df.to_dict('records')
    
    def get_leader_detail(self, anomaly_id: int):
        """查询单只股票的完整分析"""
        engine = create_engine(config_util.get_config('common.url'))
        
        with engine.connect() as conn:
            df = pd.read_sql("""
                SELECT * FROM stock_anomaly_analysis
                WHERE anomaly_id = :anomaly_id
            """, conn, params={'anomaly_id': anomaly_id})
        
        if df.empty:
            return None
        
        return df.iloc[0].to_dict()

def check_migration_needed() -> bool:
    """检查是否需要迁移数据"""
    engine = create_engine(config_util.get_config('common.url'))
    
    with engine.connect() as conn:
        # 检查旧表是否存在且有数据
        result = conn.execute(text("""
            SELECT TABLE_NAME 
            FROM INFORMATION_SCHEMA.TABLES 
            WHERE TABLE_SCHEMA = DATABASE()
              AND TABLE_NAME IN ('stock_anomaly_leader', 'stock_anomaly_leader_size')
        """))
        
        old_tables = [r[0] for r in result.fetchall()]
        
        if not old_tables:
            print("✅ 无旧表，无需迁移")
            return False
        
        # 检查数据量
        total_old = 0
        for table in old_tables:
            result = conn.execute(text(f"SELECT COUNT(*) FROM {table}"))
            count = result.fetchone()[0]
            total_old += count
            print(f"  {table}: {count}条")
        
        if total_old == 0:
            print("✅ 旧表无数据，无需迁移")
            return False
        
        print(f"⚠️ 发现旧数据共 {total_old} 条，需要迁移")
        return True
```
